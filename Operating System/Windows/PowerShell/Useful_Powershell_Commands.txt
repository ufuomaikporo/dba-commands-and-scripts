----------------------------------- like -------------------------------------------------

- Allows you to make comparisions using wildcard expressions containing the following ---->   *, ?, and [ ] 

------------------------------------- match --------------------------------------------

- Allows you to use the full list of regular expression characters.
- Evaluates as true if the string on one side of the match operator is contained anywhere within the 
	other string  (eg. "WPL_" -match "W" evaluates as true)

-- Quantifiers
*	--> Zero or more times. (used in a similar way as the '%' in SQL for -like operator)
+	--> One or more times.
?	--> Zero or one time.
{n,m} --> At least n, but no more than m times


-- Anchors
^  --> match the start of a string
$ --> match the end of a string

\w --> match any word character
\d --> match any number
.    --> match any character except a new line
[]   ---> match a single character with a character (and or number) range. You can use multiple brackets for matching more than one  
        character (neater aproach) or use a single bracket with multiple ranges within it

--------Example 1) of usage with an array 

Look for a database name that starts with 'WPL' in the 'name' property of an array called $DBList

$DBList.name -match '^WPL'

----------- Example 2) In this example '_' is a literal character included in the search

Get-ChildItem "B:\Archive\" | where-object{$_.name -match "\w*OCT_20\w*"} 
	IS EQUIVALENT TO 
Get-ChildItem "B:\Archive\" | where-object{$_.name -match "*OCT*20*"}

------------------------------------  winrar command line -------------------------------------

COMMANDS

u <-------> update archive

SWITCHES
https://documentation.help/WinRAR/HELPSwitches.htm

-m4 or -m5 <------? m5 for best compression, m4 for good compression
-ibck   <----------------------> run WinRAR in background
-ep    <------------------------------> exlude full file path in archive (archive file wihout including the parent folders)
-ep1   <------------------------------> include subfolders 
-df    <-------------------------------> delete files after archiving

------------------------------------------- Examples ----------------------------------------------------

1) #Location of the Winrar executable file on operating system

2) # Store the list of files you want to compress in a temporary file or in a powershell array variable

$BuSourceList.FullName | Out-File -FilePath 'C:\temp\temp_file.txt' -Force

#compress the list of files in the tempfile into an archive file 'ArchiveFile.rar'
& $winrar u -ibck -ep -m4 -y 'C:\Temp\ArchiveFile.rar' @'C:\temp\temp_file.txt'

OR

# Compress all the contents of a folder 'C:\Temp\' into the archive file 'ArchiveFile.rar'
& $winrar u -ibck -ep -m4 -y 'C:\Temp\ArchiveFile.rar' 'C:\Temp\*'

OR

-------------------------------------- Get the last friday of the month for a given date ----------------------------

function Get-LastFridayOfMonth([DateTime] $d) {
    $lastDay = new-object DateTime($d.Year, $d.Month, [DateTime]::DaysInMonth($d.Year, $d.Month))
    $diff = ([int] [DayOfWeek]::Friday) - ([int] $lastDay.DayOfWeek)
    if ($diff -ge 0) {
        return $lastDay.AddDays(- (7-$diff))
    }
    else
    {
        return $lastDay.AddDays($diff)
    }
}

#OR

# Choose the day of the week (0 = Sunday, 1 = Monday ...). This calculates the last specified day of the month
$DayOfWeek = 5
$CurrentDate = get-date 
$NextMonth = $CurrentDate.AddMonths(1)
$LastDayOfCurrentMonth = $NextMonth.AddDays(-$NextMonth.Day)
if ([int]$LastDayOfCurrentMonth.DayOfWeek -ge $DayOfWeek) {
    $LastDayOfWeek = $LastDayOfCurrentMonth.AddDays(-$LastDayOfCurrentMonth.DayOfWeek + $DayofWeek)
}
else {
    $SubtractDays = [int]$LastDayOfCurrentMonth.DayOfWeek + (7 - [int]$DayofWeek)
    $LastDayOfWeek = $LastDayOfCurrentMonth.AddDays(-$SubtractDays)
}

function Get-LastFridayOfMonth([DateTime] $d) {
    $lastDay = new-object DateTime($d.Year, $d.Month, [DateTime]::DaysInMonth($d.Year, $d.Month))
    $diff = ([int] [DayOfWeek]::Friday) - ([int] $lastDay.DayOfWeek)
    if ($diff -ge 0) {
        return $lastDay.AddDays(- (7-$diff))
    }
    else
    {
        return $lastDay.AddDays($diff)
    }
}

$BeforeReindex = New-Object System.Collections.ArrayList
foreach($DB in $nosession) 
{
    $IndexStats = Invoke-Sqlcmd -ServerInstance $ServerName -Database $DB.DBName -Query $GetIndexStats 
 
    $FragPercentage = ($IndexStats | Measure-Object -Property FragmentationPercentage -Average -Maximum)
    $PageCount = ($IndexStats | Measure-Object -Property page_count -Average -Maximum)

	# Create the columns of the array
    $temp = New-Object System.Object
    $temp | Add-Member -MemberType NoteProperty -Name "DBName" -Value $DB.DBName
    $temp | Add-Member -MemberType NoteProperty -Name "MaxFrag" -Value $FragPercentage.Maximum
    $temp | Add-Member -MemberType NoteProperty -Name "AvgFrag" -Value $FragPercentage.Average
    $temp | Add-Member -MemberType NoteProperty -Name "MaxPageCount" -Value $PageCount.Maximum
    $temp | Add-Member -MemberType NoteProperty -Name "AvgPageCount" -Value $PageCount.Average

	# Add a new row entry (made from columns of the array) to the array object
    $BeforeReindex.Add($temp) | Out-Null 
}

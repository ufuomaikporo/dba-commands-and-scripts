SELECT DISTINCT
    C.ENGFROMM, C.ENGTOM, e.RouteId,
    E.ROUTENAME, C.globalid as sourceid, C.CoatingType, CONVERT(varchar, C.DateApplied, 106) as DateApplied,
    C.jointcoatingtype, p.globalid as StationSeriesId
FROM gas.P_EXTERNALCOATING C 
    INNER JOIN gas.P_ENGINEERINGNETWORK E ON E.ROUTEID = C.engrouteid
    INNER JOIN gas.PIPELINELINE P ON E.ROUTEID = P.routeid
    INNER JOIN (select distinct GLOBALID, max(GDB_FROM_DATE) as MaxDate
        from gas.P_EXTERNALCOATING 
        WHERE GDB_BRANCH_ID = 0
        group by GLOBALID) G
    on C.GLOBALID = G.GLOBALID and C.GDB_FROM_DATE = G.MaxDate
WHERE
AssetGroup in (3,20,2) and lifecyclestatus = 8 AND P.GDB_BRANCH_ID = 0 and P.GDB_IS_DELETE = 0
AND legacysystem is not null AND jointlinebody = 'Line Body'

####################################### Revised Optimized Query ###########################################

with lineInfo as
(
    select distinct e.RouteId,
E.ROUTENAME,p.globalid as StationSeriesId

    FROM gas.P_ENGINEERINGNETWORK E 
INNER JOIN gas.PIPELINELINE P ON E.ROUTEID = P.routeid
WHERE AssetGroup in (3,20,2) and lifecyclestatus = 8 AND P.GDB_BRANCH_ID = 0 and P.GDB_IS_DELETE = 0
AND legacysystem is not null
)

 

select C.ENGFROMM, C.ENGTOM, I.RouteId,
I.ROUTENAME, C.globalid as sourceid, C.CoatingType, CONVERT(varchar, C.DateApplied, 106) as DateApplied,
C.jointcoatingtype, StationSeriesId 
from lineInfo I
INNER JOIN gas.P_EXTERNALCOATING C ON I.ROUTEID = C.engrouteid
INNER JOIN (
select distinct GLOBALID, max(GDB_FROM_DATE) as MaxDate
    from gas.P_EXTERNALCOATING 
    WHERE GDB_BRANCH_ID = 0
    group by GLOBALID
) G on C.GLOBALID = G.GLOBALID and C.GDB_FROM_DATE = G.MaxDate
where C.jointlinebody = 'Line Body'

--------------------SEE CONNECTED SESSIONS (users connected to databases on server)
USE master 

SELECT  @@ServerName AS SERVER
 ,d.name
 ,sp.spid
 ,sp.loginame
 ,sp.login_time
 ,sp.status
 ,sp.open_tran
 ,sp.hostname
 ,sp.program_name
 ,physical_io
 ,memusage MemoryUsage
 ,sp.blocked BlockedBy
 ,cpu 
 ,last_batch
 ,waittime
 ,lastwaittype
 ,waitresource
 ,nt_username WindowsUserName
 ,t.text
FROM sys.databases d 
LEFT JOIN sysprocesses sp ON d.database_id = sp.dbid
CROSS APPLY sys.dm_exec_sql_text(sp.sql_handle) AS t
WHERE loginame IS NOT NULL AND loginame NOT LIKE 'NT SERVICE\%'    
AND hostname <> ''
AND sp.program_name NOT IN ('Microsoft SQL Server Management Studio - Transact-SQL IntelliSense', 'Microsoft SQL Server Management Studio')
ORDER BY 2

----------------------------- kILL SESSIONS connected to a database

DECLARE @SQL NVARCHAR(MAX)
DECLARE db_cursor CURSOR FOR
SELECT 'KILL ' + CAST(spid AS NVARCHAR)
FROM sys.databases d 
LEFT JOIN sysprocesses sp ON d.database_id = sp.dbid
WHERE  name = 'DRAS_IRASv6_DEMO_QA' AND 
loginame IS NOT NULL  AND 
loginame <> 'sa' AND loginame NOT LIKE 'NT SERVICE\%'     

OPEN db_cursor  
FETCH NEXT FROM db_cursor INTO @SQL  

WHILE @@FETCH_STATUS = 0

     EXECUTE (@SQL)
	
     FETCH NEXT FROM db_cursor INTO @SQL 
END 

CLOSE db_cursor  
DEALLOCATE db_cursor 

-------------------------------------- SEE ACTIVE RUNNING SESSIONS -----------------------------

------- OPTION 1)

SELECT p.spid, p.status,p.hostname,p.loginame,p.cpu,r.start_time, t.text, p.program_name
FROM sys.dm_exec_requests AS r, sys.sysprocesses p 
CROSS APPLY sys.dm_exec_sql_text(p.sql_handle) t
WHERE p.status NOT IN ('sleeping', 'background')
AND r.session_id=p.spid

------- OPTION 2)

SELECT 
p.loginame,
p.status,
p.hostname,
req.session_id,
req.status,
req.command,
req.cpu_time,
req.total_elapsed_time,
sqltext.TEXT,
program_name
FROM sys.dm_exec_requests req, sys.sysprocesses p 
CROSS APPLY sys.dm_exec_sql_text(sql_handle) AS sqltext
WHERE  req.session_id=p.spid

-- Get the server privileges and role granted directly to user

DECLARE @UserName NVARCHAR(400) = 'EDDYFINDT\Dras users'
SELECT p1.name UserName, p1.type_desc UserType,
	  'Server Permission' PermissionType, permission_name COLLATE SQL_Latin1_General_CP1_CI_AS  AS Permission 
FROM sys.server_permissions pm
JOIN sys.server_principals p1 ON pm.grantee_principal_id = p1.principal_id
JOIN sys.server_principals p2 ON pm.grantor_principal_id = p2.principal_id
WHERE p1.name = @UserName
UNION
SELECT p2.name UserName, p2.type_desc UserType,
	  'Database Role' PermissionType, p1.name COLLATE SQL_Latin1_General_CP1_CI_AS Role
FROM sys.server_role_members rm
JOIN sys.server_principals p1
ON p1.principal_id = rm.role_principal_id
JOIN sys.server_principals p2 ON p2.principal_id = rm.member_principal_id
WHERE p2.name = @UserName

-------- OPTION 2) Only shows permissions explicitly granted to user (excludes permissions granted through roles)

SELECT all_permissions.permission_name AS [Permission Name], p.name AS [Current User]
FROM (
    SELECT DISTINCT permission_name 
    FROM sys.server_permissions
) all_permissions
LEFT JOIN (
    SELECT b.name, a.permission_name
    FROM sys.server_permissions a
    JOIN sys.server_principals b on a.grantee_principal_id = b.principal_id 
    WHERE b.name  = 'EDDYFINDT\DBA'
) p ON p.permission_name = all_permissions.permission_name

SELECT sp.state_desc AS "Permission State", sp.permission_name AS "Permission",
sl.name "Principal Name",sp.class_desc AS "Object Class", ep.name "End Point"
FROM sys.server_permissions AS sp
  JOIN sys.server_principals AS sl
    ON sp.grantee_principal_id = sl.principal_id
  LEFT JOIN sys.endpoints AS ep
    ON sp.major_id = ep.endpoint_id

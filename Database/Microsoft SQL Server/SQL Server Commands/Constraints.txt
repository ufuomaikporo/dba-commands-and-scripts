-------------------------------- FIND ALL FOREIGN KEYS REFERENCING AN OBJECT --------------------

-- BEST METHOD SO FAR to find all foreign keys referencing a column


SELECT OBJECT_NAME(fkc.constraint_object_id) AS fk_constraint,
	   SCHEMA_NAME(t1.schema_id) AS fk_table_schema,
	   OBJECT_NAME(fkc.parent_object_id) AS fk_table,
	   COL_NAME(fkc.parent_object_id, fkc.parent_column_id) AS fk_column,
	   SCHEMA_NAME(t2.schema_id) AS referenced_table_schema,
	   OBJECT_NAME(fkc.referenced_object_id) AS referenced_table,
	   COL_NAME(fkc.referenced_object_id, fkc.referenced_column_id) AS referenced_column
FROM sys.foreign_key_columns fkc
JOIN sys.tables t1
ON t1.object_id = fkc.parent_object_id
JOIN sys.tables t2 
ON t2.object_id = fkc.referenced_object_id
WHERE OBJECT_NAME(fkc.referenced_object_id) = 'Corporation'
ORDER BY 1

-- List all foreign keys referencing a specified column plus all foreign keys on the specified column referencing columns on other tables
		
SELECT *
FROM
(SELECT OBJECT_NAME(constraint_object_id) AS constraint_name,
OBJECT_NAME(parent_object_id) AS fk_table,
COL_NAME(parent_object_id, parent_column_id) AS fk_column,
OBJECT_NAME(referenced_object_id) AS pk_table,
COL_NAME(referenced_object_id, referenced_column_id) AS pk_column
FROM sys.foreign_key_columns
WHERE OBJECT_NAME(referenced_object_id) = ''' + @TableName +  '''
AND COL_NAME(referenced_object_id, referenced_column_id) = ''' + @ColumnName + '''
UNION
SELECT OBJECT_NAME(constraint_object_id) AS constraint_name,
OBJECT_NAME(parent_object_id) AS fk_table,
COL_NAME(parent_object_id, parent_column_id) AS fk_column,
OBJECT_NAME(referenced_object_id) AS pk_table,
COL_NAME(referenced_object_id, referenced_column_id) AS pk_column
FROM sys.foreign_key_columns
WHERE OBJECT_NAME(parent_object_id) = ''' + @TableName + '''
AND COL_NAME(parent_object_id, parent_column_id) = ''' + @ColumnName + '''
) SQ

---------------------------------------- Recursive List of Foreign Key dependencies ------------------------

-- Get a recursive list of all foreign key tables that depend on the InlineInspectionRange table.  May not work for all tables
	due to a maximum recursion of 100

With MyCTE
AS (
    --Anchor. Get the roots
	SELECT  1 AS Level, 
			OBJECT_NAME(referenced_object_Id) ReferencedTable, 
			OBJECT_NAME(parent_object_id) FKTable
	FROM 
	sys.foreign_key_columns 
	WHERE OBJECT_NAME(referenced_object_Id) = 'InlineInspectionRange'
    UNION ALL
    --Recursive. Get the direct descendants of the "previous" case
    SELECT  m.level + 1, OBJECT_NAME(t.referenced_object_Id), OBJECT_NAME(t.parent_object_id)
    FROM sys.foreign_key_columns t 
        INNER JOIN MyCTE m on m.FKTable = OBJECT_NAME(t.referenced_object_Id)
)
SELECT *
FROM MyCTE ORDER BY level

------------------------ Construct Command to DROP all foreign key constraints referencing a specified table

-- Disable all foreign key constraints on ListDomain Tables

SELECT 'ALTER TABLE [' + SCHEMA_NAME(t1.schema_id) + '].[' +  OBJECT_NAME(fkc.parent_object_id) + '] NOCHECK CONSTRAINT ' +
OBJECT_NAME(fkc.constraint_object_id) AS fk_constraint
FROM sys.foreign_key_columns fkc
JOIN sys.foreign_keys fk ON fkc.constraint_object_id = fk.object_id
JOIN sys.tables t1 ON t1.object_id = fkc.parent_object_id
JOIN sys.tables t2 ON t2.object_id = fkc.referenced_object_id
WHERE OBJECT_NAME(fkc.referenced_object_id) = 'ListDomain'
AND is_disabled = 0

------------------------------- Construct Command to ENABLE foreign keys referencing a list of tables 

SELECT 'ALTER TABLE [' + SCHEMA_NAME(t1.schema_id) + '].[' +  OBJECT_NAME(fkc.parent_object_id) + '] CHECK CONSTRAINT ' +
OBJECT_NAME(fkc.constraint_object_id) AS fk_constraint
FROM sys.foreign_key_columns fkc
JOIN sys.foreign_keys fk ON fkc.constraint_object_id = fk.object_id
JOIN sys.tables t1 ON t1.object_id = fkc.parent_object_id
JOIN sys.tables t2 ON t2.object_id = fkc.referenced_object_id
WHERE OBJECT_NAME(fkc.referenced_object_id) IN ('DomainCatalog', 'ListDomain', 
'DomainCatalogColXRef', 'VirtualTable', 'VirtualColumn')
AND is_disabled = 1

------------------- Construct Command to RECREATE foreign key constraints referencing a table FOR SQL Server 2017 and higher

SELECT DISTINCT 'ALTER TABLE [' + SCHEMA_NAME(tab.schema_id) +'].['+ OBJECT_NAME(fkc1.parent_object_id) 
+ '] WITH CHECK ADD CONSTRAINT [' + OBJECT_NAME(constraint_object_id) + ']' 
+ CHAR(10) + 'FOREIGN KEY(' + (SELECT '[' + STRING_AGG(COL_NAME(fkc2.parent_object_id, fkc2.parent_column_id), '], [') + ']' 
					FROM sys.foreign_key_columns fkc2 WHERE constraint_object_id = fkc1.constraint_object_id) 
+ CHAR(10) + ') REFERENCES [' + SCHEMA_NAME(tab2.schema_id) + '].[' + OBJECT_NAME(fkc1.referenced_object_id) + '](' 
+ (SELECT '[' + STRING_AGG(COL_NAME(fkc2.referenced_object_id, fkc2.referenced_column_id), '], [')  + ']'
FROM sys.foreign_key_columns fkc2 WHERE constraint_object_id = fkc1.constraint_object_id) +')' AS FK_Definition
FROM sys.foreign_key_columns fkc1
JOIN sys.tables tab
ON tab.object_id = fkc1.parent_object_id
JOIN sys.tables tab2 
ON tab2.object_id = fkc1.referenced_object_id
WHERE OBJECT_NAME(referenced_object_id) = 'FK_BUCK_COU_THREAT_EXIST'

------------------- Construct Command to RECREATE foreign key constraints referencing a table FOR SQL Server 2016 and lower

SELECT DISTINCT OBJECT_NAME(constraint_object_id) ConstraintName, 
				OBJECT_NAME(fkc.parent_object_id) ConstraintTableName, 
				OBJECT_NAME(fkc.referenced_object_id) ReferenceTableName,
				'IF OBJECT_ID (''' + OBJECT_NAME(constraint_object_id) + ''', ''F'') IS NULL
			BEGIN
			ALTER TABLE ' + QUOTENAME(SCHEMA_NAME(tab.schema_id)) +'.'+ QUOTENAME(OBJECT_NAME(fkc.parent_object_id) )
			+ ' WITH CHECK ADD CONSTRAINT ' + QUOTENAME(OBJECT_NAME(constraint_object_id)) 
			+ CHAR(10) + 'FOREIGN KEY(' + (SELECT STUFF(
														(SELECT ', ' + QUOTENAME(COL_NAME(fkc2.parent_object_id, fkc2.parent_column_id))
														 FROM sys.foreign_key_columns fkc2 
														 WHERE constraint_object_id = fkc.constraint_object_id
														 ORDER BY fkc2.constraint_column_id
														 FOR XML PATH(N''), TYPE
														 ).value(N'.[1]', N'nvarchar(max)'), 1, 2, N''
													 )
										)
			+ CHAR(10) + ') REFERENCES ' + QUOTENAME(SCHEMA_NAME(tab2.schema_id)) + '.' + QUOTENAME(OBJECT_NAME(fkc.referenced_object_id)) + '(' 
			+ (SELECT STUFF(
							(SELECT ', ' + QUOTENAME(COL_NAME(fkc2.referenced_object_id, fkc2.referenced_column_id))
							FROM sys.foreign_key_columns fkc2 
							WHERE constraint_object_id = fkc.constraint_object_id
							ORDER BY fkc2.constraint_column_id
							FOR XML PATH(N''), TYPE
							).value(N'.[1]', N'nvarchar(max)'), 1, 2, N''
						  )
				)
				+')
			END ' AS ConstraintDefinition
FROM sys.foreign_key_columns fkc
JOIN sys.tables tab
ON tab.object_id = fkc.parent_object_id
JOIN sys.tables tab2 
ON tab2.object_id = fkc.referenced_object_id
WHERE OBJECT_NAME(fkc.constraint_object_id) = 'FK_RCHCAStructureId2'

---------------- ALTERNATIVE (without function calls)

SELECT DISTINCT ct.name FKTable, pt.name ParentTable, fk.name ConstraintName,
'IF OBJECT_ID( '''+ fk.name +''', ''F'') IS NULL BEGIN ALTER TABLE [' + scc.name +'].['+ ct.name 
+ '] WITH CHECK ADD CONSTRAINT [' + fk.name + ']' 
+ CHAR(10) + 'FOREIGN KEY(' + (SELECT STUFF(
											(SELECT ', ' + QUOTENAME(c2.name)
												FROM sys.foreign_key_columns fkc2 
												JOIN sys.columns c2 ON c2.object_id = fkc2.parent_object_id
												AND c2.column_id = fkc2.parent_column_id
												WHERE constraint_object_id = fkc.constraint_object_id
												ORDER BY fkc2.constraint_column_id
												FOR XML PATH(N''), TYPE
												).value(N'.[1]', N'nvarchar(max)'), 1, 2, N''
											)
										)
			+ CHAR(10) + ') REFERENCES ' + QUOTENAME(scp.name) + '.' + QUOTENAME(pt.name) + '(' 
			+ (SELECT STUFF(
							(SELECT ', ' + QUOTENAME(c2.name)
							FROM sys.foreign_key_columns fkc2 
							JOIN sys.columns c2 ON c2.object_id = fkc2.referenced_object_id
							AND c2.column_id = fkc2.referenced_column_id
							WHERE constraint_object_id = fkc.constraint_object_id
							ORDER BY fkc2.constraint_column_id
							FOR XML PATH(N''), TYPE
							).value(N'.[1]', N'nvarchar(max)'), 1, 2, N''
						  )
				)
				+')
			END ' AS FK_Definition
FROM sys.foreign_keys fk
JOIN sys.foreign_key_columns fkc ON fk.object_id = fkc.constraint_object_id
JOIN sys.tables ct
ON ct.object_id = fkc.parent_object_id
JOIN sys.tables pt 
ON pt.object_id = fkc.referenced_object_id
JOIN sys.columns cc ON cc.object_id = fkc.parent_object_id AND fkc.parent_column_id = cc.column_id
JOIN sys.columns pc ON pc.object_id = fkc.referenced_object_id AND fkc.referenced_column_id = pc.column_id
JOIN sys.schemas scp ON scp.schema_id = pt.schema_id
JOIN sys.schemas scc ON scc.schema_id = ct.schema_id
WHERE fk.name = 'FK_RCStrucScVertexStrucId'

NOTE**: Only 1 clustered index allowed per table

-- Find primary key, unique key and foreign key indexes on a table

SELECT i.name, 
	   OBJECT_NAME(ic.object_id) AS TableName, 
	   COL_NAME(ic.object_id, column_id) As columnName,
	   i.type_desc,
	   is_unique,
	   CASE WHEN is_primary_key = 1 THEN 'Primary Key'
			WHEN is_unique_constraint = 1 THEN 'Unique Key'
			ELSE 'Nonclustered Index'
		END AS 'Constraint Type'
FROM sys.indexes i
JOIN sys.index_columns ic
ON i.object_id = ic.object_id
AND i.index_id = ic.index_id
JOIN sys.tables tab
ON tab.object_id = i.object_id
WHERE OBJECT_NAME(ic.object_id) = 'Variable'
ORDER BY name

----------------------------------- Recreate primary, unique, and any other indexes on table --------------------

SELECT t.name AS TableName, 
c.name ColumnName, 
i.name ConstraintName, 
CASE WHEN is_primary_key = 1 THEN 'PK' WHEN is_unique_constraint = 1 THEN 'UQ' ELSE 'Index' END AS IndexType
, CASE WHEN is_primary_key = 1 OR is_unique_constraint = 1 THEN 
'ALTER TABLE dbo.' + t.name
+ ' WITH CHECK ADD CONSTRAINT ' 
+ QUOTENAME(i.name)
+ CASE WHEN is_primary_key = 1 THEN ' PRIMARY KEY ' WHEN is_unique_constraint = 1 THEN ' UNIQUE ' END + i.type_desc COLLATE SQL_Latin1_General_CP1_CI_AS 
ELSE 'CREATE ' + i.type_desc COLLATE SQL_Latin1_General_CP1_CI_AS + ' INDEX ' + QUOTENAME(i.name) + ' ON ' + QUOTENAME(sc.name)
+ '.' + QUOTENAME(t.name)  END + ' ('
+ (SELECT STUFF(
				(SELECT ', ' + QUOTENAME(c2.name) +
					CASE WHEN is_descending_key = 1 THEN ' DESC'
						ELSE ' ASC'
					END
				FROM sys.index_columns ic2
				JOIN sys.columns c2 ON c2.column_id = ic2.column_id
				AND c2.object_id = ic2.object_id
				WHERE ic2.object_id = ic.object_id
				AND ic2.index_id = ic.index_id
				AND ic2.is_included_column = 0
				ORDER BY key_ordinal
				FOR XML PATH(N''), TYPE
				).value(N'.[1]', N'nvarchar(max)'), 1, 2, N''
			) 
			
	) + ')' + CASE WHEN EXISTS (SELECT * FROM sys.index_columns 
	WHERE object_id = i.object_id AND index_id = i.index_id AND is_included_column = 1) THEN
			' INCLUDE (' + COALESCE((SELECT STUFF(
				(SELECT ', ' + QUOTENAME(c2.name)
				FROM sys.index_columns ic2
				JOIN sys.columns c2 ON c2.column_id = ic2.column_id
				AND c2.object_id = ic2.object_id
				WHERE ic2.object_id = ic.object_id
				AND ic2.index_id = ic.index_id
				AND ic2.is_included_column = 1
				ORDER BY key_ordinal
				FOR XML PATH('')
				), 1, 2, ''
			)), '') + ')' ELSE '' END +  ' WITH (PAD_INDEX =' + CASE WHEN is_padded = 0 THEN ' OFF' ELSE ' ON' END 
			+ ', STATISTICS_NORECOMPUTE =' + CASE WHEN s.no_recompute = 0 THEN ' OFF' ELSE ' ON' END 
			+ ', ALLOW_ROW_LOCKS =' + CASE WHEN allow_row_locks = 0 THEN ' OFF' ELSE ' ON' END 
			+ ', ALLOW_PAGE_LOCKS =' + CASE WHEN allow_page_locks = 0 THEN ' OFF' ELSE ' ON' END + ')'   AS Index_Definition
FROM IRQ_IRASV6_LINK.sys.indexes i
JOIN IRQ_IRASV6_LINK.sys.index_columns ic
ON i.object_id = ic.object_id
AND i.index_id = ic.index_id
JOIN IRQ_IRASV6_LINK.sys.stats s ON  s.object_id = i.object_id AND s.name = i.name
JOIN IRQ_IRASV6_LINK.sys.tables t ON i.object_id = t.object_id
JOIN IRQ_IRASV6_LINK.sys.columns c ON c.object_id = ic.object_id AND c.column_id = ic.column_id
JOIN IRQ_IRASV6_LINK.sys.schemas sc ON sc.schema_id = t.schema_id
WHERE t.name = 'VTGUIDLookup'


----------------------------------- Recreate primary and unique keys on a table -------------------------------

SELECT DISTINCT 'ALTER TABLE ' + QUOTENAME(SCHEMA_NAME(t.schema_id)) + '.' + QUOTENAME(OBJECT_NAME(ic.object_id)) 
+ ' WITH CHECK ADD CONSTRAINT ' 
+ QUOTENAME(i.name)
+ CASE WHEN is_primary_key = 1 THEN ' PRIMARY KEY ' WHEN is_unique_constraint = 1 THEN ' UNIQUE ' END 
+ i.type_desc COLLATE SQL_Latin1_General_CP1_CI_AS  + ' ('
+ (SELECT STUFF(
				(SELECT ', ' + QUOTENAME(COL_NAME(object_id, column_id)) +
					CASE WHEN is_descending_key = 1 THEN ' DESC'
						ELSE ' ASC'
					END
				FROM sys.index_columns 
				WHERE object_id = ic.object_id
					AND index_id = ic.index_id
				ORDER BY key_ordinal
				FOR XML PATH(N''), TYPE
				).value(N'.[1]', N'nvarchar(max)'), 1, 2, N''
			)
			
	) + ') WITH (PAD_INDEX =' + CASE WHEN is_padded = 0 THEN ' OFF' ELSE ' ON' END 
			+ ', STATISTICS_NORECOMPUTE =' + CASE WHEN s.no_recompute = 0 THEN ' OFF' ELSE ' ON' END 
			+ ', ALLOW_ROW_LOCKS =' + CASE WHEN allow_row_locks = 0 THEN ' OFF' ELSE ' ON' END 
			+ ', ALLOW_PAGE_LOCKS =' + CASE WHEN allow_page_locks = 0 THEN ' OFF' ELSE ' ON' END + ')'
FROM sys.indexes i
JOIN sys.index_columns ic
ON i.object_id = ic.object_id
AND i.index_id = ic.index_id
JOIN sys.stats s ON  s.object_id = i.object_id AND s.name = i.name
JOIN sys.tables t ON i.object_id = t.object_id
WHERE (i.is_primary_key = 1 OR i.is_unique_constraint = 1)
AND OBJECT_NAME(i.object_id) = 'CalcOutflowStatisticsType'

------------------------------- Recreate non-primary and non-unique Indexes on a table -------------------------

SELECT 'CREATE ' + i.type_desc COLLATE SQL_Latin1_General_CP1_CI_AS + ' INDEX ' + QUOTENAME(i.name) + ' ON ' + QUOTENAME(SCHEMA_NAME(t.schema_id) )
+ '.' + QUOTENAME(OBJECT_NAME(t.object_id)) 
+ '(' + (SELECT STUFF(
				(SELECT ', ' + QUOTENAME(COL_NAME(object_id, column_id)) + CASE WHEN is_descending_key = 1 THEN ' DESC ' ELSE ' ASC ' END
				FROM sys.index_columns ic1
				WHERE i.object_id = ic1.object_id
				AND i.index_id = ic1.index_id
				AND is_included_column = 0
				ORDER BY key_ordinal
				FOR XML PATH('')
				), 1, 2, ''
			)) + ')' + CASE WHEN EXISTS (SELECT * FROM sys.index_columns WHERE object_id = i.object_id AND index_id = i.index_id AND is_included_column = 1) THEN
			' INCLUDE (' + COALESCE((SELECT STUFF(
				(SELECT ', ' +  QUOTENAME(COL_NAME(object_id, column_id))
				FROM sys.index_columns ic1
				WHERE i.object_id = ic1.object_id
				AND i.index_id = ic1.index_id
				AND is_included_column = 1
				ORDER BY key_ordinal
				FOR XML PATH('')
				), 1, 2, ''
			)), '') + ')' ELSE '' END + ' WITH (PAD_INDEX =' + CASE WHEN is_padded = 0 THEN ' OFF' ELSE ' ON' END 
			+ ', STATISTICS_NORECOMPUTE =' + CASE WHEN s.no_recompute = 0 THEN ' OFF' ELSE ' ON' END 
			+ ', ALLOW_ROW_LOCKS =' + CASE WHEN allow_row_locks = 0 THEN ' OFF' ELSE ' ON' END 
			+ ', ALLOW_PAGE_LOCKS =' + CASE WHEN allow_page_locks = 0 THEN ' OFF' ELSE ' ON' END + ')'
FROM sys.indexes i
JOIN sys.tables t ON i.object_id = t.object_id
JOIN sys.stats s ON  s.object_id = i.object_id AND s.name = i.name
WHERE is_primary_key = 0 AND is_unique_constraint = 0

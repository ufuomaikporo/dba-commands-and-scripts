------------------------------------------- COALSECE/STRING_AGG ---------------------------------------------

-------- OPTION 1) STRING_AGG

STRING_AGG function only works on SQL Server 2017 and higher

-- The WITHIN GROUP command allows you to sort the data
SELECT STRING_AGG(name, ', ') WITHIN GROUP (ORDER BY column_id)
FROM sys.columns
WHERE is_computed = 0 AND OBJECT_NAME(object_id) = 'ListDomain'


----- OPTION 2) STUFF

------- ALTERNATIVE TO STRING_AGG FOR SQL Server 2016 and lower

-- forms a comma-delimited list of names of the columns from the ListDomain table 
SELECT STUFF(
				(SELECT ', ' + name
				FROM sys.columns 
				WHERE OBJECT_NAME(object_id) = 'ListDomain'
				ORDER BY column_id
				FOR XML PATH(N''), TYPE
				).value(N'.[1]', N'nvarchar(max)'), 1, 2, N''
			)

OR -- the only difference is what is written after 'FOR XML PATH...' the results are the same otherwise

SELECT STUFF(
				(SELECT ', ' + name
				FROM sys.columns 
				WHERE OBJECT_NAME(object_id) = 'ListDomain'
				ORDER BY name
				FOR XML PATH('')
				), 1, 2, ''
			)

---------- OPTOIN 3) COALESCE

-- The example of the COALESCE function below can be done with the STRING_AGG function 

---------- a) the seperator (',') is placed after the first table name and then after every other table name 

DECLARE @ColumnList NVARCHAR(MAX)

SELECT TOP 5 @ColumnList = COALESCE(@ColumnList, '') + name + ', '
FROM sys.tables

---------- b) IF @ColumnList is NULL, the seperator (',') is placed after the first table name and then after every other table 
			except for the table in the list. Otherwise the seperator (',') will be placed before the first table name than after 
			every other table name except for the last table in the list

SELECT TOP 5 @ColumnList = COALESCE(@ColumnList + ', ', '') + name 
FROM sys.tables

-- column name is stored in the @col_name variable which is combined into the qualified column name stored in the

SET @q_column_name = @table_name + '.' + @col_name

# WINDOWS COMMANDS 
#################################################################################################

xp_cmdshell is a built-in Microsoft SQL Server extended stored procedure that allows users to run 
Windows command prompt commands from within SQL. It can be used to view files in a windows directory,
create zip files, etc.

- It needs to be enabled first by doing the following

-- 1) Show Advanced Options
EXEC sp_configure 'show advanced options', 1
GO
RECONFIGURE
GO

-- 2) Enable xp_cmdshell
EXEC sp_configure 'xp_cmdshell', 1
GO
RECONFIGURE
GO

-- 3) Hide Advanced Options
EXEC sp_configure 'show advanced options', 0
GO
RECONFIGURE
GO

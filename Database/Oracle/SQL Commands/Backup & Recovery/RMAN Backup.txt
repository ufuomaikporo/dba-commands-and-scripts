# RMAN BACKUPS
###################################################################################

NOTES**: You can make full, full-level-0, incremental-level-1 and cummulative-level-1 backups for 
database, datafile and tablespace backups

LEVEL 0 backup: is a base level full backup to be used as part of a incremental backup strategy 
cmd: BACKUP AS [COPY | COMPRESSED BACKUPSET] INCREMENTAL LEVEL 0 
	 TAG 'tag'
	 [DATABASE | DATAFILE | TABLESPACE tablespace_name1, tablespace_name2]

LEVEL 1 differential: is a backup on top of a LEVEL 0 backup that captures any changes since the last LEVEL 0 or LEVEL 1 backup
cmd: BACKUP INCREMENTAL LEVEL 1 
	 TAG 'tag'
	 [DATABASE | DATAFILE | TABLESPACE tablespace_name1, tablespace_name2]

LEVEL 1 cummulative: is a backup on top of a LEVEL 0 backup that captures any changes since the last LEVEL 0, for the purpose of
					speeding up recovery as opposed to recoverying with multiple differential backups and a LEVEL 0, you can recovery 
					with 1 cummulative and the LEVEL 0.
cmd: BACKUP INCREMENTAL LEVEL 1 CUMMULATIVE 
	 TAG 'tag'
	 [DATABASE | DATAFILE | TABLESPACE tablespace_name1, tablespace_name2]

-- Check backup history in RMAN catalog

COL status FORMAT A10
COL hrs FORMAT 999.99

SELECT session_key, 
	   input_type, status,
	   to_char(START_TIME,'mm/dd/yy hh24:mi') start_time,
	   to_char(END_TIME,'mm/dd/yy hh24:mi') end_time,
	   elapsed_seconds/3600 hrs 
FROM V$RMAN_BACKUP_JOB_DETAILS
ORDER BY session_key;

Incremental Backups
###################################################################################

-------------------------------Level 0 (base) backup------------------------------

-- can specify 'AS COPY' for image copy, other is a backup set

BACKUP INCREMENTAL LEVEL 0 DATABASE | TABLESPACE tablespace_name | DATAFILE #filenumber;

--------------------------Level 1 differnetial backup---------------------------------

-- level 1 backups are always backup sets
-- differential backup is a backup of changed data blocks since last 
    level 0 or 1 backup
-- faster backup than cummulative level 1 backup, but with slower recovery

BACKUP INCREMENTAL LEVEL 1 DATABASE | TABLESPACE tablespace_name | DATAFILE #filenumber;

---------------------------Level 1 cumulative backup-----------------------------

-- larger and slower backup than differential backup, but with a faster recovery

BACKUP INCREMENTAL LEVEL 1 CUMULATIVE DATABASE | TABLESPACE tablespace_name | DATAFILE #filenumber;

----------------------------------------------- BACKUP COMMAND PARAMETERS -------------------------------------------

-- CHECK LOGICAL checks for logical corruption
-- PLUS ARCHIVELOG does the following
	1) Runs the ALTER SYSTEM ARCHIVE LOG CURRENT statement.
	2) Runs BACKUP ARCHIVELOG ALL. If backup optimization is enabled, then RMAN skips logs that it has already backed up to the specified device.
	3) Backs up the rest of the files specified in the BACKUP command.
	4) Runs the ALTER SYSTEM ARCHIVE LOG CURRENT statement.
	5) Backs up any remaining archived logs generated during the backup. If backup optimization is not enabled, then RMAN backs up the logs generated in Step 1 plus all the logs generated during the backup.
-- DELETE ALL INPUT deletes each backed-up archived redo log file from all log archiving destinations.

----------------------------------------------- INCREMENTAL LEVEL 0 (Base Full Backup)BACKUP COMMAND ---------------------------------------------
https://docs.oracle.com/cd/B19306_01/backup.102/b14192/bkup004.htm
https://docs.oracle.com/en/database/oracle/oracle-database/21/bradv/rman-backup-concepts.html#GUID-0E273817-1EE0-40FD-BD92-51C4A0CAAF79
https://docs.oracle.com/en/database/oracle/oracle-database/21/bradv/rman-backup-concepts.html#GUID-0E273817-1EE0-40FD-BD92-51C4A0CAAF79

RUN
{
	# Set backup location and filename for controlfile and spfile backup within the run-block 
	SET CONTROLFILE AUTOBACKUP FORMAT FOR DEVICE TYPE DISK TO 'B:\Backup\ControlFile_n_SPFile_%F';

	# Allocate 4 channels for parallel processes to run operations within the run-block. and set the location and file name for backupsets
	ALLOCATE CHANNEL ch1 DEVICE TYPE DISK FORMAT = 'B:\Fast Recovery Area\ORA19C22\Backups\%d_%M_%D_%Y_%U' MAXPIECESIZE 10G;
	ALLOCATE CHANNEL ch2 DEVICE TYPE DISK FORMAT = 'B:\Fast Recovery Area\ORA19C22\Backups\%d_%M_%D_%Y_%U' MAXPIECESIZE 10G;
	ALLOCATE CHANNEL ch3 DEVICE TYPE DISK FORMAT = 'B:\Fast Recovery Area\ORA19C22\Backups\%d_%M_%D_%Y_%U' MAXPIECESIZE 10G;
	ALLOCATE CHANNEL ch4 DEVICE TYPE DISK FORMAT = 'B:\Fast Recovery Area\ORA19C22\Backups\%d_%M_%D_%Y_%U' MAXPIECESIZE 10G;
	
	# Backup the tablespaces listed and the archive log files, then delete the archive log files (based on archivelog deletion policy)
	# from all archive log locations. Add a tag to both the archive log backup and database backup
	BACKUP AS COMPRESSED BACKUPSET 
	INCREMENTAL LEVEL 0
	TABLESPACE Users, Index TAG 'Full_backup'
	PLUS ARCHIVELOG
	DELETE ALL INPUT TAG 'Full_backup';
	
	# release the processing channels 
	RELEASE CHANNEL ch1;
	RELEASE CHANNEL ch2;
	RELEASE CHANNEL ch3;
	RELEASE CHANNEL ch4;
}

RUN
{
	SET CONTROLFILE AUTOBACKUP FORMAT FOR DEVICE TYPE DISK TO 'B:\NightlyBackup\ControlFile_n_SPFile_%F.BKP';

	COPIES 3 FORMAT 'B:\NightlyBackup\%d_%M_%D_%Y_%U.BKP',

	VALIDATE CHECK LOGICAL TABLESPACE Tablespace1, Tablespace2 ;  # check that 
	RELEASE CHANNEL ch1;
	RELEASE CHANNEL ch2;
	RELEASE CHANNEL ch3;
	RELEASE CHANNEL ch4;

-------------------------------------- INCREMENTAL LEVEL 1 (CUMULATIVE BACKUP) ------------------------------------

-- Backup all changes since last level 0 (takes longer and takes up more space than a differential backup)

RUN
{
	# Set backup location and filename for controlfile and spfile backup within the run-block 
	SET CONTROLFILE AUTOBACKUP FORMAT FOR DEVICE TYPE DISK TO 'B:\Backup\ControlFile_n_SPFile_%F';

	# Allocate 4 channels for parallel processes to run operations within the run-block. and set the location and file name for backupsets
	ALLOCATE CHANNEL ch1 DEVICE TYPE DISK FORMAT = 'B:\Fast Recovery Area\ORA19C22\Backups\%d_%M_%D_%Y_%U' MAXPIECESIZE 10G;
	ALLOCATE CHANNEL ch2 DEVICE TYPE DISK FORMAT = 'B:\Fast Recovery Area\ORA19C22\Backups\%d_%M_%D_%Y_%U' MAXPIECESIZE 10G;
	ALLOCATE CHANNEL ch3 DEVICE TYPE DISK FORMAT = 'B:\Fast Recovery Area\ORA19C22\Backups\%d_%M_%D_%Y_%U' MAXPIECESIZE 10G;
	ALLOCATE CHANNEL ch4 DEVICE TYPE DISK FORMAT = 'B:\Fast Recovery Area\ORA19C22\Backups\%d_%M_%D_%Y_%U' MAXPIECESIZE 10G;

	# Backup the tablespaces listed and the archive log files, then delete the archive log files (based on archivelog deletion policy)
	# from all archive log locations. Add a tag to both the archive log backup and database backup
	BACKUP AS COMPRESSED BACKUPSET 
	INCREMENTAL LEVEL 1 CUMULATIVE
	TABLESPACE Users, Index TAG 'Cumulative_Backup'             
	PLUS ARCHIVELOG
	DELETE ALL INPUT TAG 'Cumulative_Backup';

	#check for physical and logical corruption within the tablespaces listed
	VALIDATE CHECK LOGICAL TABLESPACE Users, Index;

	# release the processing channels 
	RELEASE CHANNEL ch1;
	RELEASE CHANNEL ch2;
	RELEASE CHANNEL ch3;
	RELEASE CHANNEL ch4;
}

------------------------------------------- INCREMENTALLY BACKUP STRATEGY COMMAND ----------------------------

HOW IT WORKS: Creates a level 0 image copy full backup the first time it runs, creates a level 1 differential backup the second time it runs
	on the 3rd time going forwared it uses the previous level 1 backup to update/roll-forward the image copy backup 
	then performs a new level 1 differential backup

-- Image copy backups take up more space than compressed backupsets so make sure there is enough space in the location chosen.
	It's best to make the image copy backup then convert it into compressed backupset to tape
	
1) -- The following provides fast recoverability within 24 hours if the script is ran daily

RUN
{
	ALLOCATE CHANNEL c1 DEVICE TYPE DISK FORMAT '\\backups.cgy.dynamicrisk.net\DB_Backups_Nightly\ORA19C\Nightly\Test\%d_%I_%M_%D_%Y_%U' MAXPIECESIZE 3G;
	ALLOCATE CHANNEL c2 DEVICE TYPE DISK FORMAT '\\backups.cgy.dynamicrisk.net\DB_Backups_Nightly\ORA19C\Nightly\Test\%d_%I_%M_%D_%Y_%U' MAXPIECESIZE 3G;
	ALLOCATE CHANNEL c3 DEVICE TYPE DISK FORMAT '\\backups.cgy.dynamicrisk.net\DB_Backups_Nightly\ORA19C\Nightly\Test\%d_%I_%M_%D_%Y_%U' MAXPIECESIZE 3G;
	ALLOCATE CHANNEL c4 DEVICE TYPE DISK FORMAT '\\backups.cgy.dynamicrisk.net\DB_Backups_Nightly\ORA19C\Nightly\Test\%d_%I_%M_%D_%Y_%U' MAXPIECESIZE 3G;

	RECOVER COPY OF DATABASE 
	WITH TAG 'incr_update';
	
	BACKUP 
	INCREMENTAL LEVEL 1 
	CHECK LOGICAL 
	FOR RECOVER OF COPY WITH TAG 'incr_update'
	DATABASE PLUS ARCHIVELOG
	DELETE ALL INPUT;
	
	RELEASE CHANNEL c1;
	RELEASE CHANNEL c2;
	RELEASE CHANNEL c3;
	RELEASE CHANNEL c4;
}

------------------------------------------------------- ALLOCATE TO SPECIFIC TASKS ------------------------------------

RMAN> RUN
{
  ALLOCATE CHANNEL ch1 DEVICE TYPE DISK FORMAT = '/backup/df/%U';
  ALLOCATE CHANNEL ch2 DEVICE TYPE DISK FORMAT = '/backup/cf/%U';
  ALLOCATE CHANNEL ch3 DEVICE TYPE sbt;
  BACKUP AS BACKUPSET # all output files are in backup sets
    # channel ch1 backs up datafiles to /backup/df directory
    DATAFILE 1,2,3,4 
    CHANNEL ch1
    # channel ch2 backs up control file copy to /backup/cf directory
    CONTROLFILECOPY '/tmp/control01.ctl'
    CHANNEL ch2;
  BACKUP AS BACKUPSET
    # channel ch3 backs up archived redo logs to tape
    ARCHIVELOG FROM TIME 'SYSDATE-14'
    CHANNEL ch3;
}

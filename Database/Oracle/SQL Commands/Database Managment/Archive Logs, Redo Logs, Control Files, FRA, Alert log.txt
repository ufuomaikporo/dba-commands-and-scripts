---------------------------- MONITOR LOG SWITCH FREQUENCY -------------------------

- Oracle Recommends 1 log switch every 15-30 minutes or less than 6 log switches per hour at peak times (high DML activity)

- Get number of log switches for each hour of each day where a log switch occured 

COL day FORMAT a15;
COL hour FORMAT a4;
COL total FORMAT 999;

SELECT
   TO_CHAR(first_time,'yyyy-mm-dd') day,
   TO_CHAR(first_time,'hh24')       hour,
   COUNT(*)                         total
FROM
   v$log_history
GROUP BY
   TO_CHAR(first_time,'yyyy-mm-dd'),TO_CHAR(first_time,'hh24')
ORDER BY
   TO_CHAR(first_time,'yyyy-mm-dd'),TO_CHAR(first_time,'hh24')
ASC;

               - allows recovery to last commited transaction

               - database must be closed for full or incremental backups (cold backup)

######################################################################################
SET FLASH RECOVERY AREA (FRA)
########################################################################################

-- FRA is space set aside on disk that can be used for archived logs, backups, flashback logs, multiplexed 
   control files, and multiplexed redo logs
-- location should be seperate from database data files, primary online redo log files
   and control files
-- size of FRA should be atleast twice the size of database to hold one backup and several 
   archived logs (Oracle Server automatically manages FRA storage based on 
                 the set retention policy)
-- WARNING** The database instance will pause/hang if there is not enough space to create archived logs

------------------------------------------------- Query FRA info --------------------------------------------

OPTION 1) Get the Location and space info on FRA

SHOW PARAMETER db_recovery_file_dest

OR 

SHOW PARAMETER db_recovery

OPTION 2)

BREAK ON REPORT
COMPUTE SUM OF percentage_space_used ON REPORT
COMPUTE SUM OF percentage_space_reclaimable ON REPORT

SELECT * FROM v$recovery_area_usage;

OPTION 3)

COL name FORMAT A30
SET LIN 200

SELECT name, 
	   space_limit/POWER(1024,3) AS quota_gb,
       ROUND(space_used/POWER(1024,3), 2) AS used_space_gb,
       ROUND((space_limit - space_used)/POWER(1024,3), 2) AS remaining_gb,
       space_reclaimable/POWER(1024,3) AS reclaimable_gb,
	   DECODE(nvl(space_used,0),0,0,round((space_used/space_limit) * 100)) pct_used,
       number_of_files AS files
FROM v$recovery_file_dest;

---------------------------------- View location info on FRA -----------------------------------------

-- view current location, disk quota, space in use, space reclaimable by deleting files,
  and total number of files in the following dynamic performance view

v$recovery_file_dest

-- view percentage of disk quota used by different files

v$recovery_area_usage

-- The following commands set the location and size of the FRA (Specifed folder location must already exist) 

ALTER SYSTEM SET db_recovery_file_dest_size = 200M;
ALTER SYSTEM SET db_recovery_file_dest = 'C:\app\Oracle\recovery_area\';

-- QUERY the V$RECOVERY_FILE_DEST view to vew information in the FRA. 
name: FRA name with location string
space_limit: disk limit specified in DB_RECOVERY_FILE_DEST_SIZE parameter
space_used: disk space used by FRA in bytes
space_reclaimable: space that can be reclaimed by deleting obolete, redundant, and other 
         low priority files through space management algorithm
number_of_files: number of files in FRA

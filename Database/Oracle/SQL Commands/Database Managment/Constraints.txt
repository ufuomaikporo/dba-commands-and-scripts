----------------FIND ALL REFERENTIAL CONSTRAINTS ON TABLES (ref_table_name)-----------------

-- Find all foreign keys that are referencing a specified table
SET PAGESIZE 10000 LINES 500
COLUMN fk_constraint FORMAT A30
COLUMN fk_table FORMAT A30
COLUMN fk_column FORMAT A30
COLUMN ref_table FORMAT A30
COLUMN ref_column FORMAT A30

SELECT uc.constraint_name AS fk_constraint,
	   child.table_name AS fk_table,
	   child.column_name AS fk_column,
	   parent.table_name AS ref_table,
	   parent.column_name AS ref_column
FROM user_constraints uc
JOIN user_cons_columns parent
	ON uc.r_constraint_name = parent.constraint_name
JOIN user_cons_columns child
	ON uc.constraint_name = child.constraint_name
WHERE parent.table_name = 'ListDomain' COLLATE BINARY_CI
	AND parent.column_name = 'Id' COLLATE BINARY_CI;

-------------------------DISABLE/DROP CONSTRAINTS ON A TABLE----------------------------------

ENABLE VALIDATE is the same as ENABLE. The constraint is checked and is guaranteed to hold for all rows.
ENABLE NOVALIDATE means the constraint is checked for new or modified rows, but existing data may violate the constraint.
DISABLE NOVALIDATE is the same as DISABLE. The constraint is not checked so data may violate the constraint.
DISABLE VALIDATE means the constraint is not checked but disallows any modification of the constrained column

-- Allow inserts of rows that violate the constraint using the following

ALTER TABLE course
DISABLE NOVALIDATE CONSTRAINT fk_course_to_course;

-- Enables the constraint for incoming inserts and validates current rows in the table
   to ensure that it complies with the constraint

ALTER TABLE course
ENABLE VALIDATE CONSTRAINT fk_course_to_course;

-- Delete a constraint

ALTER TABLE course DROP CONSTRAINT fk_course_to_course;

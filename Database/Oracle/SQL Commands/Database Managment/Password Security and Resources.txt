Managing Profiles

-- Possible password settings for profiles

FAILED_LOGIN_ATTEMPTS: number of failed login attempts before the account is locked
PASSWORD_LIFE_TIME: Limits the number of days the same password can be used for authentication
PASSWORD_REUSE_TIME: number of days before an old password can be reused
PASSWORD_REUSE_MAX: number of password changes before the same password can be reused
PASSWORD_VERIFY_FUNCTION: Function that determines the level of password complexity
PASSWORD_LOCK_TIME: number of days an account will be locked after specified consecutive 
                   failed login attempts
PASSWORD_GRACE_TIME: number of days after a password expiration that you are still allowed 
                     to connect to the database. Password can't be used after grace period
                     ends
INACTIVE_ACCOUNT_TIME: number of consecutive days without a log in, after which the account
                       will be locked  

-- Possible resource limits

CPU_PER_SESSION: Total amount of CPU time measured in hundredths of seconds
SESSIONS_PER_USER: Maximum number of concurrent sessions allowed for a user
CONNECT_TIME: maximum connect time measured in minutes
IDLE_TIME: maximum inactive time measured in minutes (not affected by long queries)
LOGICAL_READS_PER_SESSION: Number of data blocks (physical and logical reads) allowed per
                             session
PRIVATE_SGA: Private space in the SGA measured in bytes (for Shared Server only) 
CPU_PER_CALL: CPU time per call in hundredths of seconds
LOGICAL_READS_PER_CALL: Number of data blocks that can be read per call

-- Create a profile with password and resource limits

CREATE PROFILE registrar LIMIT
FAILED_LOGIN_ATTEMPTS            10
PASSWORD_LIFE_TIME               180
PASSWORD_REUSE_TIME              60
PASSWORD_REUSE_MAX               3
PASSWORD_VERIFY_FUNCTION         ORA12C_VERIFY_FUNCTION
PASSWORD_LOCK_TIME               1
PASSWORD_GRACE_TIME              7
SESSIONS_PER_USER                1
IDLE_TIME                        20;

--Modifying the password settings for a profile 

ALTER PROFILE default LIMIT
FAILED_LOGIN_ATTEMPTS            10
PASSWORD_LIFE_TIME               180
PASSWORD_REUSE_TIME              UNLIMITED
PASSWORD_REUSE_MAX               UNLIMITED
PASSWORD_VERIFY_FUNCTION         NULL
PASSWORD_LOCK_TIME               1
PASSWORD_GRACE_TIME              7
INACTIVE_ACCOUNT_TIME            UNLIMITED;

--TO UNLOCK A LOCKED ACCOUNT

ALTER USER jeff ACCOUNT UNLOCK;

-----------------------------Password verification function---------------

-- You can use the describe command against a password verify function that exists
   in the database

desc ORA12_VERIFY_FUNCTION

-- Or you can search for in the function in the dba_objects table

SET PAGES 999 LINES 120
COL object_name FORMAT A30

SELECT object_name, status
FROM dba_objects
WHERE object_name LIKE '%VERIFY%';

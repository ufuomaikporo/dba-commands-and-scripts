----------------------------Creating Permanent Tables-----------------------------

Tips:

-- Define nullable columns at the of your create table column definition list
-- Best to specify the tablespace
-- if rows are going to change in size PCTFREE can be set to 0. But if rows will grow in
   size use the formula (100*average_bytes_added_in_updates/average_row_length) to determine
   the value of PCTFREE for a table

-- The following creates a table in the users tablespace leaving 30% of the space
   within data blocks in the segment for updates

CREATE TABLE Employee (
	EEID  Number(7) NOT NULL,
	Fname CHAR(25),
 	Lname CHAR(25))
	TABLESPACE Users
	PCTFREE 30;

-- Create table in users tablespace with the primary key index in the index_tbs
   tablespace

CREATE TABLE Sample (
ID NUMBER(5),
CONSTRAINT SamplePK PRIMARY KEY (ID)) Tablespace users
ENABLE PRIMARY KEY USING INDEX TABLESPACE index_tbs;

-- Creates a table with a primary key and unique indexes in an index_tbs
   tablespace while the table resides in the 'data' tablespace

CREATE TABLE Sample (
ID NUMBER(5),
SIN NUMBER(9),
CONSTRAINT SamplePK PRIMARY KEY (ID),
CONSTRAINT SinUnique UNIQUE(SIN))
TABLESPACE data
ENABLE PRIMARY KEY USING INDEX TABLESPACE index_tbs
ENABLE CONSTRAINT SinUnique USING INDEX TABLESPACE index_tbs;

-- The following example creates a table with a primary key index in 
   a 'user_indx' tablespace. The table is stored in the user_data tablespace
   and a storage clause is specifed for the table to override the tablespace 
   storage settings. all constraints are defined at the column level

CREATE TABLE HR.DEPT2 (
DEPT_ID number (4) CONSTRAINT dept2_dept_id_pk PRIMARY KEY
	USING INDEX STORAGE (INITIAL 100K NEXT 100K PCTINCREASE 20) TABLESPACE USER_INDX,
DEPT_NAME varchar2(30) CONSTRAINT dept2_dept_name_nn NOT NULL)
STORAGE (INITIAL 100K NEXT 100K PCTINCREASE 20)
PCTFREE 10 PCTUSED 60
TABLESPACE USER_DATA;

-- Another example. All constraints defined at the column level

CREATE TABLE HR.EMP2 
(
EMP_ID number (4) CONSTRAINT emp2_emp_id_pk PRIMARY KEY
	USING INDEX STORAGE (INITIAL 100K NEXT 100K PCTINCREASE 0) TABLESPACE USER_INDX,
LAST_NAME varchar2(30) CONSTRAINT emp2_last_name_nn NOT NULL,
FIRST_NAME varchar2(30) CONSTRAINT emp2_first_name_nn NOT NULL,
SALARY number(7) CONSTRAINT emp2_salary_ck CHECK (SALARY>=0),
DEPT_ID number(4) CONSTRAINT emp2_dept2_dept_id_fk REFERENCES hr.dept2(dept_id)
)
STORAGE (INITIAL 100K NEXT 100K PCTINCREASE 0)
PCTFREE 20 PCTUSED 60
TABLESPACE USER_DATA;

-------------------------------------------- Create a Global Temporary Tables------------------------------

-- Global temporary tables have a persistent structure but impersistent data. That means the 
   data is deleted either at the end of each  transaction (i.e. after a COMMIT)or when the session ends. 
	You can use ON COMMIT PRESERVE ROWS (the default is ON COMMIT DELETE ROWS) to preserve the table 
	data until the session is terminated or until you explictly truncate the table.
	You have to explicitly drop a global temp table for it to be removed from the database;

CREATE GLOBAL TEMPORARY TABLE supplier_temp 
(supplier_id number(8), 
 company varchar2(80), 
 rep varchar2(80)
) ON COMMIT PRESERVE ROWS;

- To drop a global temporary table, you must first truncate the table

TRUNCATE TABLE supplier_temp;
DROP TABLE supplier_temp;

----------------------------------------- CREATE PRIVATE TEMPORARY TABLES --------------------------------------

- Private temporary tables have impersistent data and structure. The table and structure is deleted either
	at the end of each transaction by default (ON COMMIT DELETE DEFINITION) or at the end of the session
	with the 'ON COMMIT PRESERVE DEFINITION' clause
	
- The table name must begin with the prefix defined in the PRIVATE_TEMP_TABLE_PREFIX initialisation parameter. The default is "ORA$PTT_".

CREATE PRIVATE TEMPORARY TABLE ora$ptt_my_temp_table (
  id           NUMBER,
  description  VARCHAR2(20)
)
ON COMMIT PRESERVE DEFINITION;

-- Modify table column definition. Only the elements mentioned in the ALTER statement are changed
   every other part of the table definition before the ALTER statement that is not mentioned stays the same
NOTE** In some instances you can change column datatype with the statement below in others the column must be empty to change column datatype

ALTER TABLE ListDomain MODIFY Id NUMBER(10,0) DEFAULT 0;

-- Re-arrange column order

Example 
1) Add new column 'c'
2) make the columns that you want to come after the new column invisible and then make them visible
	ALTER TABLE t MODIFY (d INVISIBLE, e INVISIBLE);
	ALTER TABLE t MODIFY (d VISIBLE, e VISIBLE);
	
If you make all columns invisible in the same statement all columns being made visible will appear in the order they
were in before being made invisible. If you want a specific order you may have to use more than one alter statement

-- Delete rows from a table (always use a where clause unless if you wan to delete the entire table data)

DELETE FROM table_name
WHERE column = value;

-- Truncating a table resets it to how it was before any data was in the table
TRUNCATE TABLE table_name

-- Dropping a table (CASCADE CONSTRAINTS removes all constraints like foreign keys and PURGE removes the deleted table from recycle bin)

DROP TABLE TableName [CASCADE CONSTRAINTS][PURGE];

-- Dropping table column

ALTER TABLE DROP COLUMN CASCADE CONSTRAINTS;

-----------------------Get number of rows in a schema table------------------------------- 

-- Query to get the size and number of rows in each table in the KNOWLEDGE schema

-- First execute the following procedure to get current statistics on the schema

EXEC DBMS_STATS.gather_table_stats('KNOWLEDGE');

-- Then run the query

COL tablespace_name FORMAT A15
COL table_name FORMAT A20
COL owner FORMAT A15

SELECT table_name,
       ds.tablespace_name,
       NVL(bytes/1024, 0) size_kb,
       NVL(num_rows, 0) num_rows
FROM user_tables dt 
  JOIN dba_segments ds
 ON (table_name = segment_name)
WHERE ds.owner = 'KNOWLEDGE'
UNION
SELECT table_name, 
       tablespace_name, 
       0,
       NVL(num_rows, 0) num_rows
FROM user_tables
WHERE num_rows = 0
ORDER BY 1;

----------------------------------------- Correlated Update --------------------------------------------

-- this is similar to performing a JOIN on 'table1' and 'table2' on columns  ----> tab1_column = tab2_column 
   then updating the 'update_column' of 'table1' with the value from the 'expression' column of 'table2' based on the 
   JOIN condition
    
UPDATE table1 alias1
SET update_column = (SELECT expression
			  FROM table2 alias2
			  WHERE alias1.tab1_column = alias2.tab2_column);

/* Trigger that updates the LastModByUserID, and LastModDateTime row-tracking columns whenever 
   any row of the table SimCalcFeatureLeak is updated. :NEW and :OLD are psuedotables similar
   to the DELETED AND INSERTED tables in SQL Server.
*/

CREATE OR REPLACE TRIGGER tu_SimCalcFeatureLeak 
   BEFORE UPDATE OR DELETE
   ON SimCalcFeatureLeak
   FOR EACH ROW
DECLARE
BEGIN
	DBMS_OUTPUT.PUT_LINE ('The OLD values are: ' || :OLD.LastModByUserID || ' and ' || :OLD.LastModDateTime);
	DBMS_OUTPUT.PUT_LINE ('The NEW value are: ' || :NEW.LastModByUserID || ' and ' || :NEW.LastModDateTime);

	-- Update the LastModByUserID and LastModDateTime columns in SimCalcFeatureLeak table with the following values
	:NEW.LastModByUserID := SYS_CONTEXT('userenv','os_user');
	:NEW.LastModDateTime := SYSDATE;
END;
/

-- Enable/disable trigger

ALTER TRIGGER tu_SimCalcFeatureTotal ENABLE/DISABLE

ALTER TABLE inventory
    ENABLE/DISABLE ALL TRIGGERS;

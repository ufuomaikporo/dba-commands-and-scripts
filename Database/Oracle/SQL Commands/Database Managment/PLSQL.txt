   r_sal   emp.sal%TYPE);

OPTION 3) -- Define a table variable using the data structure of the rows of an existing table (i.e. the dept table/cursor)

                INDEX BY BINARY_INTEGER;

OPTION 4) -- Define a table of record type

TYPE emp_record_type IS RECORD
  (r_ename emp.ename%TYPE,
   r_sal   emp.sal%TYPE);
   
TYPE DDL_TABLE_TYPE IS TABLE OF emp_record_type
                INDEX BY BINARY_INTEGER;
				
tmpListDomainForeignKeys DDL_TABLE_TYPE;

------------------------------------- TABLE VARIABLE METHODS -------------------------------------

EXISTS(n)   Returns TRUE if nth element of the table exists.

COUNT       The number of elements (rows) in the plsql table

FIRST       First and Last index no.s in the table
LAST        returns NULL if table is empty

PRIOR(n)    Returns index no that preceeds n in the plsql table

NEXT(n)     Returns index no that succeeds n in the plsql table

EXTEND(n,i) Append n copies of the 'i'th element to a plsql table
            i defaults to NULL n defaults to 1

TRIM(n)     Remove n elements from the end of a plsql table
            n defaults to 1

DELETE(m,n) Delete elements in range m...n
            m defaults to = n
            n defaults to ALL elements
			
-- Get the length (number of rows) of the table array	
		
tmpListDomainForeignKeys.COUNT;

-- Delete entire tables

tmpListDomainForeignKeys.DELETE;

tmpListDomainForeignKeys(1).r_ename

	item := tmpListDomainForeignKeys(i) -- for table of a single row [OPTION 1) and 2) in table variable declaration]
END LOOOP;

LOOPS AND CURSORS
###################################################################################

-------------------------------------------Cursors---------------------------------

-- You can use cursors to loop through the rows in a table. A cursor automatically 
   stores the rows of the query within it's definition into varibles defined as, 
   cursor_name. column_name

-- Defining a cursor. A cursor can be defined with a query. The cursor definition must be in DECLARE block

CURSOR emp_cursor IS
    SELECT ename, sal
    FROM emp
    WHERE sal IS NOT NULL
    ORDER BY sal DESC;
	
-------------------------------------------- LOOPS -----------------

-- For-Loop with cursor. emp_rec is a variable local to the for-loop

FOR emp_rec IN emp_cursor LOOP
    employee := emp_rec.ename;
	salary := emp_rec.sal;
    v_counter := v_counter + 1;
	
	-- You could also populate the PLSQL 'salary' variable like below
	SELECT sal INTO salary
	FROM emp
	WHERE ename = emp_rec.ename;
	
END LOOP;
  

-- you can create a generic for-loop as follows. You can use this loop  to iterate through each row in a table variable by replacing the 
	'5' with a variable that stores the number of rows in the table. The 
	statement for loop excution below is execute as long as 1 <= i <= 5. You can
	also add an 'EXIT WHEN' command to exit the loop

 FOR i IN 1..5 LOOP
 	EXECUTE IMMEDIATE tmpListDomainForeignKeys(i);
 END LOOP;

                (ActionItemRevisionId, ActionItemPropertyId, PropertyValue)

                FROM ActionItemRevision
                WHERE NOT EXISTS (SELECT 1 FROM ActionItemPropertyValue 
									WHERE ActionItemRevisionId = Id 
									AND ActionItemPropertyId = PropertyId)]';

-- The following example uses dynamic SQL to create a sequence using the result
   of a query (the max client id from the client table) as the starting value
   for the sequence. NOTICE** there is no semi-colon at the end of the dynamically generated command

DECLARE
  ddl_stmt VARCHAR2(200);
BEGIN
  SELECT 'CREATE SEQUENCE client_id_seq START WITH ' || MAX(client_id) || ' INCREMENT BY 1'
  INTO ddl_stmt
  FROM client;

  EXECUTE IMMEDIATE ddl_stmt;
END;

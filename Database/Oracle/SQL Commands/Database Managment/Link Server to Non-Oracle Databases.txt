----------------------------------------- Create Or Add linked server for SQL Server database ----------------

STEP 1) Download Oracle Database Gataways and run setup installer (leave no spaces in file folder names)
- select ODBC for non-Oracle database and or ODB for Microsoft SQL Server

STEP 2) Customize the initialization parameter file located at ---> ORACLE_HOME\dg4msql\admin\initdg4msql.ora
The content of the file should be as follows. The Database link needs to be dropped and recreated every time a change
 is made to the initdg4msql.ora file

******************************************************

# This is a customized agent init file that contains the HS parameters
# that are needed for the Database Gateway for Microsoft SQL Server

#
# HS init parameters
#
HS_FDS_CONNECT_INFO=HostName:Port   #This option connects you to the master database since no database name is specified
# for named instance ---> HS_FDS_CONNECT_INFO = Hostname/InstanceName/DatabaseName 
# For default instance ---> HS_FDS_CONNECT_INFO = Hostname:port//DatabaseName
HS_FDS_TRACE_LEVEL=OFF
HS_FDS_RECOVERY_ACCOUNT=RECOVER
HS_FDS_RECOVERY_PWD=RECOVER

*******************************************************

STEP 3) Setup Listener.ora file. Enter SID_DESC entry for dg4msql and restart the listener after making changes

************************************************************************

SID_LIST_LISTENER =
  (SID_LIST =
    (SID_DESC =
      (SID_NAME = CLRExtProc)
      (ORACLE_HOME = C:\App\Oracle\product\19.0.0\db_home_1)
      (PROGRAM = extproc)
      (ENVS = "EXTPROC_DLLS=ONLY:C:\App\Oracle\product\19.0.0\db_home_1\bin\oraclr19.dll")  
    )
    (SID_DESC=
	(SID_NAME=dg4msql)
	(ORACLE_HOME = C:\App\Oracle\product\19.0.0\db_home_1)
	(PROGRAM=dg4msql)
    )
  )

LISTENER =
  (DESCRIPTION_LIST =
    (DESCRIPTION =
      (ADDRESS = (PROTOCOL = TCP)(HOST = localhost)(PORT = 1521))
      (ADDRESS = (PROTOCOL = IPC)(KEY = EXTPROC1521))
    )
  )


************************************************************************

STEP 4) Setup TNSnames.ora file

*******************************************************************************

dg4msql =
  (DESCRIPTION =
    (ADDRESS = (PROTOCOL = TCP)(HOST = localhost)(PORT = 1521))
    (CONNECT_DATA = (SID = dg4msql))
      (HS = OK)
  )
  
**************************************************************************

STEP 5) Create a database link  

CREATE PUBLIC DATABASE LINK DBLinkName CONNECT TO
 "user" IDENTIFIED BY "password" USING 'tns_name_entry';
 
-- Deleting Database link

DROP PUBLIC DATABASE LINK SQL2016_AU_IRASv6_STAGE;
 

STEP 6) Test the database link. it should return an 'X'

SELECT * FROM DUAL@DBLinkName;

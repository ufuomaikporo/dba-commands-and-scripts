----------------------------------------- CASE EXPRESSION -----------------------------

-- The example below is for a simple case expression. The charater set of the comparison value in each WHEN statement is
in varchar2 by default. if the searched value (mls.material) is not in the same character set (i.e it is NCHAR, NVARCHAR, etc) then it
needs to be converted using the CAST function for the query to work. Searched-case expressions will implicitly perform the conversion

Select
CASE CAST(mls.material AS CHAR(10))
    WHEN 'X42' THEN 289
    WHEN 'X46' THEN 317
    WHEN 'X52' THEN 359
    WHEN 'X56' THEN 386
    WHEN 'X60' THEN 413
    WHEN 'X65' THEN 448
    WHEN 'X70' THEN 482
    WHEN 'X80' THEN 551
    END as PipeGrade
from P_MAINLINE_SEGMENTS mls
join PIPELINELINE pl on pl.name = mls.pl_name
where MLS.assetgroup in (3)

-- Searched case expression

Select
CASE 
    WHEN mls.material = 'X42' THEN 289
    WHEN mls.material = 'X46' THEN 317
    WHEN mls.material = 'X52' THEN 359
    WHEN mls.material = 'X56' THEN 386
    WHEN mls.material = 'X60' THEN 413
    WHEN mls.material = 'X65' THEN 448
    WHEN mls.material = 'X70' THEN 482
    WHEN mls.material = 'X80' THEN 551
    END as PipeGrade
from P_MAINLINE_SEGMENTS mls
join PIPELINELINE pl on pl.name = mls.pl_name
where MLS.assetgroup in (3)

----------------------------- SEARCH AND REPLACE STRING -----------------------------

-- The function below is the same as function in SQL Server 
REPLACE (Source, Old_string, new_string)

- For regular expression search and replace use the function below
search_string: You can use '|' as 'or' symbol to search for multiple different characters at once
postion parameter : is the index of the source string where the search should begin
occurrence: if you specify 0, Oracle replaces all occurrences of the match if you specify n > 0
        Oracle replaces the nth occurrence
match_parameter: 'i' (include single quotes) specifies case-insensitive match, 'c' specifies case-sensitive match

https://www.techonthenet.com/oracle/functions/regexp_replace.php

REGEXP_REPLACE(source_string, search_string, replace_string,position,occurrence,match_parameter)

-- The example below replaces all periods, dashes, underscores, and spaces in 'DomainDescr' column with no-space
NOTE**: You must use '\' to escape special characters such as '.', '&', etc

REGEXP_REPLACE(DomainDescr, ' |-|_|\.', '')

---------------------------------------- COMPARISON ------------------------------------------------

-- comparison with regular expression --> https://www.oracletutorial.com/oracle-string-functions/oracle-regexp_like/

REGEXP_LIKE(source_string, search_pattern [, match_parameter]);

Example: Notice that '%' wildcard match is not used. But if needed you can use '.*' to match any number of characters except null

SELECT df.tablespace_name, 
       df.file_name
FROM dba_data_files df
WHERE NOT REGEXP_LIKE(tablespace_name, 'UNDO|SYSTEM|SYSAUX|USER')
ORDER BY 3 DESC;

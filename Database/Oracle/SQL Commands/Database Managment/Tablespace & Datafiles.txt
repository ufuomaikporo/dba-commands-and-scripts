----------------ALTERNATIVE (MORE ACCURATE) QUERY TO GET SIZE OF DATAFILES AND UNUSED SPACE ('Savings')

-- new_val directive causes value from column 'value' to be stored in variable 'blksize'

SET PAGES 1000
SET LIN 200
SET VERIFY OFF
COLUMN file_name FORMAT A65
COLUMN UsedMB FORMAT 999,999.99 HEADING "Smallest|Size|Poss.(MB)"
COLUMN CurrentSizeMB FORMAT 999,999 HEADING "Current|Size(MB)"
COLUMN MaxSize FORMAT A15 HEADING "Max|Size(MB)"
COLUMN FreeSpace  FORMAT 999,999 HEADING "Unused|Space(MB)"
COLUMN freePerc FORMAT 999 HEADING "Free|Space(%)"
COLUMN value NEW_VAL blksize
SELECT value FROM v$parameter WHERE name = 'db_block_size';

SELECT file_name,
       ROUND((nvl(hwm,1)*&&blksize)/1024/1024, 2) UsedMB,
       ROUND(blocks*&&blksize/1024/1024, 2) CurrentSizeMB,
       CASE
			WHEN ROUND(maxbytes/1024/1024) >= 33554431 THEN 'UNLIMITED'
			ELSE TO_CHAR(ROUND(maxbytes/1024/1024), '999,999')
	   END MaxSize,
       ROUND( blocks*&&blksize - nvl(hwm,1)*&&blksize)/1024/1024 FreeSpace,
	   ROUND(( blocks*&&blksize - nvl(hwm,1)*&&blksize )/(blocks*&&blksize) * 100, 2)freePerc
FROM dba_data_files a,
     ( SELECT file_id, max(block_id+blocks-1) hwm
         FROM dba_extents
        GROUP by file_id ) b
WHERE a.file_id = b.file_id(+)
UNION
SELECT file_name,
       ROUND( (nvl(hwm,1)*&&blksize)/1024/1024, 2) UsedMB,
       ROUND( blocks*&&blksize/1024/1024, 2) CurrentSizeMB,
       CASE
			WHEN ROUND(maxbytes/1024/1024) >= 33554431 THEN 'UNLIMITED'
			ELSE TO_CHAR(ROUND(maxbytes/1024/1024), '999,999')
	   END MaxSize,
       ROUND( blocks*&&blksize - nvl(hwm,1)*&&blksize)/1024/1024 FreeSpace,
	   ROUND(( blocks*&&blksize - nvl(hwm,1)*&&blksize )/(blocks*&&blksize) * 100, 2)freePerc
FROM dba_temp_files a,
     ( SELECT file_id, max(block_id+blocks-1) hwm
         FROM dba_extents
        GROUP by file_id ) b
WHERE a.file_id = b.file_id(+)
ORDER BY 3 DESC;

------------------------ COPY NON-ORACLE MAINTAINED TABLESPACE DEFINITIONS --------------

SELECT 'CREATE '|| CASE WHEN dt.bigfile = 'YES' THEN 'BIGFILE ' END|| 'TABLESPACE ' || df.tablespace_name || CHR(10) ||
'DATAFILE '''|| df.file_name|| '''' || CHR(10) ||
'SIZE ' ||
CASE 
	WHEN CEIL(COALESCE(seg.UsedSpace_mb, 0)/1024/1024) < 5 THEN 5
	ELSE CEIL(COALESCE(seg.UsedSpace_mb, 0)/1024/1024)
END
|| 'M ' ||
CASE
	WHEN SUM(maxbytes)/1000/1024 = 0 THEN NULL
	WHEN SUM(maxbytes)/1000/1024 < 33554.416 
		THEN 'AUTOEXTEND ON NEXT ' || 
			ROUND((SUM(maxbytes)/1024/1024)*0.2)||
			'M MAXSIZE ' || SUM(maxbytes)/1024/1024 || 'M '
	WHEN SUM(maxbytes)/1000/1024 >= 33554.416 
		THEN 'AUTOEXTEND ON NEXT ' || 
		CASE 
			WHEN ROUND(SUM(df.bytes)/1024/1024*0.3) < 10 THEN 10
			ELSE ROUND(SUM(df.bytes)/1024/1024*0.3)
		END|| 'M MAXSIZE UNLIMITED' 
	ELSE NULL
END	
|| CHR(10) ||
'EXTENT MANAGEMENT '|| dt.extent_management || 
CASE WHEN dt.allocation_type = 'SYSTEM' THEN ' AUTOALLOCATE'
	WHEN dt.allocation_type = 'UNIFORM' THEN ' UNIFORM SIZE '|| dt.initial_extent/1024/1024 || 'M'
END ||	CHR(10) ||
'SEGMENT SPACE MANAGEMENT ' || dt.segment_space_management AS Create_Tablespace_ddl
FROM dba_data_files df
LEFT JOIN dba_tablespaces dt
ON df.tablespace_name = dt.tablespace_name
LEFT JOIN (SELECT tablespace_name, sum(bytes) AS UsedSpace_mb
			FROM dba_segments
			GROUP BY tablespace_name) seg
	ON seg.tablespace_name = df.tablespace_name
WHERE NOT REGEXP_LIKE (df.tablespace_name, 'SYSTEM|SYSAUX|USERS|UNDO')
AND df.tablespace_name like 'ETP%' OR df.tablespace_name = 'TEST'
GROUP BY df.tablespace_name, 
         dt.bigfile,
		 df.file_name, 
		 dt.segment_space_management, 
		 dt.extent_management,
		 dt.initial_extent, 
		 dt.allocation_type,
		 seg.usedSpace_mb;

Local and Dictionary Manged Tablespaces

-- You can create bigfile or smallfile (default) tablespace. smallfile tablespaces have a datafile size limit 

REM-------------CREATING A DICTIONARY MANAGED TABLESPACE---------------------------------

CREATE TABLESPACE test
DATAFILE 'C:\CHAPTERS\C_DRIVE\data\test.dbf' SIZE 5M
AUTOEXTEND ON NEXT 2M MAXSIZE 10M
MINIMUM EXTENT 200K                        
EXTENT MANAGEMENT DICTIONARY
DEFAULT STORAGE 
 (INITIAL 200K                                   
  NEXT 200K
  PCTINCREASE 0
  MINEXTENTS 2
  MAXEXTENTS 25);

-- minimum extent size allocated to an object (only applies to dictionary managed tablespaces)
-- INITAL 200K is the size of the first extent allocated to an object
-- NEXT 200K is the size of the next extent allocated
-- PCTINCREASE is the percent increase in size of each subsequent extent after the second extent is allocated (on third extent and so forth)
REM--------cannot specify PCTINCREASE for rollback segments. Rollback segments have a PCTINCREASE of 0
-- MINEXTENTS is the minimum number of extents allocated when an object is created (only applies to dictionary managed tablespaces)
-- MAXEXTENTS is the maximum number of extents allocated for objects (only applies to dictionary managed tablespaces)

REM----------------------CREATING LOCALLY MANGED TABLESPACES----------------------------

-- SEGMENT SPACE MANAGEMENT AUTO sets up a tablespace for Automatic Segment Space Management 
   (ASSM). This maintains the segment space allocations info within the object's segement
   itself and manages the pctfree and pctused values automatically. Setting it to MANUAL
   maintains the segment space allocation info in the data dictionary. You can set pctfree
   and pctused in this case. 

CREATE TABLESPACE test
DATAFILE 'C:\CHAPTERS\C_DRIVE\data\test.dbf' SIZE 5M
AUTOEXTEND ON NEXT 2M MAXSIZE 10M
EXTENT MANAGEMENT LOCAL AUTOALLOCATE
SEGMENT SPACE MANAGEMENT AUTO;

CREATE TABLESPACE test
DATAFILE 'C:\CHAPTERS\C_DRIVE\data\test.dbf' SIZE 5M REUSE
AUTOEXTEND ON NEXT 2M MAXSIZE 10M
EXTENT MANAGEMENT LOCAL UNIFORM SIZE 2m
SEGMENT SPACE MANAGEMENT AUTO; -- AUTO is the default

CREATE TABLESPACE stocks 
DATAFILE 'C:\CHAPTERS\C_DRIVE\datastock1.dbf' SIZE 10M, 
         'C:\CHAPTERS\C_DRIVE\datastock2.dbf' SIZE 10M,
         'C:\CHAPTERS\C_DRIVE\datastock3.dbf' SIZE 10M; 

--UNIFORM specifies uniform extent allocation (uniform in size)
--AUTOALLOCATE allows the database to choose the extent sizes which may not be uniform (DEFAULT)
--SEGMENT SPACE MANAGEMNT only applies to permanent, locally managed tablespaces
--EXTENT MANAGEMENT is always LOCAL for locally managed tablespaces
--REUSE specifies that the datafile should be overwritten if it already exists
REM-----Lets you specify whether Oracle Database should track the used 
REM------and free space in the tablespace segments using free lists or bitmaps
REM------AUTO is for bitmap (recommended)
REM------MANUAL is for free lists (need to specify PCTUSED, FREELIST and FREELIST GROUPS in this case)

-------------------------Taking a tablespace offline--------------------------------------

-- datafiles that are not in the SYSTEM or UNDO tablspace can be taken offline to allow
  you to open the database or pdb while the datafile is being restored 

WARNING:
If you must take a tablespace offline, use the normal (OFFLINE NORMAL) option (the default) if possible; 
this guarantees that the tablespace will not require recovery to come back online, even if you 
reset the redo log sequence (using an ALTER DATABASE OPEN RESETLOGS statement after incomplete media recovery) 
before bringing the tablespace back online.  

ALTER TABLESPACE tbs2 OFFLINE IMMEDIATE;

------------------------------------- Offline Move (Older method verions before 12c) ----------------------------

1) ALTER TABLESPACE tablespaceName OFFLINE;
2) Move/rename the files in the OS
	linux > mv 'oldFilePath' 'newFilePath'
	windows > copy 'oldFilePath' 'newFilePath'
	windows > rename 'oldFilePath' 'newFilePath' 
3) ALTER DATABASE RENAME FILE 'oldFilePath' TO 'newFilePath'
4) ALTER TABLESPACE tablespaceName ONLINE;

OR

1) shutdown the database ------> SHUTDOWN IMMEDIATE
2) start the database in mount mode ------> STARTUP MOUNT
3) move the files to the new locations in the OS
4) Rename the datafile ---> ALTER DATABASE RENAME FILE '/oracle1/dbs/db3.f' TO '/oracle2/dbs/db3.f';
5) ALTER DATABASE OPEN

# SHRINK DATAFILES

SET VERIFY OFF
SET PAGES 1000
SET LIN 200
COLUMN file_name FORMAT a60 WORD_WRAPPED
COLUMN UsedMB FORMAT 99,999,999.99 HEADING "Smallest|Size|Poss.(MB)"
COLUMN CurrentSizeMB FORMAT 99,999,999.99 HEADING "Current|Size(MB)"
COLUMN FreeSpace  FORMAT 99,999,999 HEADING "Poss.|Savings(MB)"
COLUMN freePerc FORMAT A15 HEADING "Free|Percentage"
BREAK ON report

-- STEP 1: To Check Database block size

COLUMN value new_val blksize
SELECT value FROM v$parameter WHERE name = 'db_block_size';


-- STEP 2: Check how much space can be reclaimed 

SELECT file_name,
       ROUND( (nvl(hwm,1)*&&blksize)/1024/1024, 2) UsedMB,
       ROUND( blocks*&&blksize/1024/1024, 2) CurrentSizeMB,
       ROUND( blocks*&&blksize - nvl(hwm,1)*&&blksize)/1024/1024 FreeSpace,
	   TO_CHAR(ROUND(( blocks*&&blksize - nvl(hwm,1)*&&blksize )/(blocks*&&blksize) * 100)) || '%' freePerc
FROM dba_data_files a,
     ( SELECT file_id, max(block_id+blocks-1) hwm
         FROM dba_extents
        GROUP by file_id ) b
WHERE a.file_id = b.file_id(+)
ORDER BY 3 DESC;

-- You can also add condition if needs to be checked for a specific tablespace

-- STEP 3: Script to reclaim unused space from the datafiles 

SET pages 10000
SET lines 300
COLUMN cmd FORMAT A300 WORD_WRAPPED

SELECT 'ALTER DATABASE DATAFILE '''||file_name||''' RESIZE ' || 
		CASE 
			WHEN CEIL(nvl(hwm,1)*&&blksize/1024/1024) > 70 THEN CEIL(nvl(hwm,1)*&&blksize/1024/1024)
			ELSE 70
		END
		|| 'M;' cmd
FROM dba_data_files a,
     ( SELECT file_id, max(block_id+blocks-1) hwm
         FROM dba_extents
        GROUP by file_id ) b
WHERE a.file_id = b.file_id(+)
	AND ROUND( blocks*&&blksize - nvl(hwm,1)*&&blksize )/(blocks*&&blksize) * 100 > 5
	AND (hwm IS NOT NULL AND nvl(hwm,1)*&&blksize/1024/1024 > 8)
	AND NOT REGEXP_LIKE(a.tablespace_name, 'UNDO|SYSTEM|SYSAUX|USER')
		ORDER BY hwm DESC;

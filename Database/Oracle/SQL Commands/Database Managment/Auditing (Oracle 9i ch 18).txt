-- Check if auditing is enabled by checking the audit parameters. If the audit_trail
   parameter is set to DB then auditing is enabled and the audit records are directed to 
   the database audit trail (SYS.AUD$), except for records that are always written to 
   the operating system audit trail
-- if audit_sys_operations is set to true then actions performed by the sys user or
   any users with the SYSDBA or SYSOPER privileges will be audited
-- if audit_syslog_level is set to true then On UNIX systems, it writes the SYS and 
   standard OS audit records to the system audit log using the SYSLOG utility.

-----------------------VIEWING UNIFIED AUDIT POLICIES ON SCHEMA OBJECTS----------------------------------

-- Report on auditing policies belonging to a schema 

SET LIN 120
SET PAGESIZE 100
COLUMN policy_name FORMAT A20
COLUMN audit_option FORMAT A20
COLUMN object_schema FORMAT A20
COLUMN object_name FORMAT A20

SELECT policy_name, audit_option, object_schema, object_name
FROM audit_unified_policies
WHERE object_schema = 'KPS'
ORDER BY object_name;

------------------------------VIEW ENABLED UNIFIED AUDIT POLICIES-------------

COLUMN policy_name FORMAT A20
COLUMN user_name FORMAT A20
COLUMN success FORMAT A8
COLUMN failure FORMAT A8

SELECT user_name, policy_name, enabled_opt, enabled_option, success, failure
FROM audit_unified_enabled_policies
WHERE policy_name = 'AUDIT_OBJPRIV_GRADE';

-- TO enable a unified audit policy for all users

AUDIT POLICY table_pol;

-- TO enable a unified audit policy only for the users hr and sh

AUDIT POLICY dml_pol BY hr, sh;

-- Disable audit policy for all users

NOAUDIT POLICY table_pol;

---------------------CREATING AUDIT POLICY ON OBJECT PRIVILEGES---------------------
CREATE AUDIT POLICY audit_objpriv_kps
ACTIONS delete on KPS.KPS_ADDRESS, 
        update on KPS.KPS_ADDRESS,
        delete on KPS.KPS_INCIDENT,
        update on KPS.KPS_INCIDENT,
        delete on KPS.KPS_PERSON,
        update on KPS.KPS_PERSON,
        delete on KPS.KPS_PERSON_ROLE,
        update on KPS.KPS_PERSON_ROLE,
        delete on KPS.KPS_POLICE, 
        update on KPS.KPS_POLICE,
	delete on KPS.KPS_POLICE_ROLE, 
        update on KPS.KPS_POLICE_ROLE,
        delete on KPS.KPS_VEHICLE,
        update on KPS.KPS_VEHICLE,
	delete on KPS.KPS_VEHICLE_ROLE,
        update on KPS.KPS_VEHICLE_ROLE

-- read audit records in unified_audit_trail data dictionary view

SET PAGES 999 LINES 300
COLUMN unified_audit_policies FORMAT A25
COLUMN object_name FORMAT A15
COLUMN action FORMAT A12
COLUMN object_schema FORMAT A15
COLUMN dbusername FORMAT A10
COLUMN os_username FORMAT A12
COLUMN event_date FORMAT A25

SELECT unified_audit_policies, 
       action_name AS action, 
       dbusername, 
       os_username,
       object_name, 
       object_schema,
       TO_CHAR(event_timestamp, 'YYYY-MON-DD HH24:MI:SS') AS event_date
FROM unified_audit_trail
WHERE unified_audit_policies = 'AUDIT_OBJPRIV_GRADE';

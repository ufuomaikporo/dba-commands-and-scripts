####################################################################################
Backups
##################################################################################

---------------------Physical backup of a tablespace---------------------------------

STEP 1) Create a directory for backups

mkdir /u04/oradata/backup/tst1

STEP 2) prepare a tablespace for backup with the following command

alter tablespace users begin backup;

STEP 3) In the Linux command prompt enter the copy command (or use "host" key word
        or "!" to enter the copy "cp" command in sql command prompt) to copy the 
        database files that are being backed up to the backup location 

cp -p /u05/oradata/tst1/users01.dbf /u04/oradata/backup/tst1/users01.dbf

STEP 4) Remove the tablespace from backup mode

alter tablespace users end backup;

-----------------------Logical backup of (oracle datapump)----------------------------

-- A logical backup is performed for the schema Disney in this example

STEP 1) Create a directory on Oracle database for the dump file

Create or replace directory DATA_PUMP_DIR as '/u04/oradata/backup/tst1';

STEP 2) Create a parameter file

dumpfile=expdp_disney_schema_tst1.dmp
logfile=expdp_disney_schema_tst1.log
schemas=(DISNEY)

STEP 3) Use the expdp oracle export utility to perform an export. The Disney schema
      is exported in this example. You can enter "/ as sysdba" to perform export as
      as SYS user

expdp "/ as sysdba" parfile=expdb_knowledge_schema_tst2.par

-------------------Import

STEP 4) Create an import parameter file (can use the export parameter file as a copy)
        them modify the import parameter file as needed using a text editor

cp expdp_disney_schema_tst1.par impdp_disney_schema_tst1.par

STEP 5) Use the import utility to perform the import

impdp parfile=impdp_disney_schema_tst1.par

-------------------------Control file backup------------------------

-- backup up the control file to trace (usually in background dump destination)

 alter database backup controlfile to trace;

###############################################################################
RMAN
#################################################################################

-- Set default location of RMAN backups enter the following command in RMAN.
   The "%U" is a file name format specifier

CONFIGURE CHANNEL DEVICE TYPE DISK FORMAT'/u04/oradata/backup/tst2/RMAN/%U';

-- Set control file autobackup location

CONFIGURE CONTROLFILE AUTOBACKUP FORMAT FOR DEVICE TYPE DISK TO '/u04/oradata/backup/tst2/RMAN/%F';

-- Configure pdb snapshot location 

CONFIGURE SNAPSHOT CONTROLFILE NAME TO '/u04/oradata/backup/tst2/RMAN/snapcf_tst1.f';

-- A summary of all database backups that occured

LIST BACKUP OF DATABASE SUMMARY;

-- A list all archive logs that were backup up 

LIST BACKUP OF ARCHIVELOG ALL;

-- A list of the controlfile backups

LIST BACKUP OF CONTROLFILE;

##################################################################################
Performance
##################################################################################

-- show how many chained and migrated rows each table has

SELECT table_name, chain_cnt 
FROM dba_tables 
WHERE chain_cnt > 0;

-- To see which rows are chained use the commands below. The ANALYZE command will
   put the rows into the CHAINED_ROWS table which is created by the 
   utlchain.sql script (in $ORACLE_HOME/rdbms/admin)

ANALYZE TABLE tablename LIST CHAINED ROWS;

SELECT * FROM chained_rows;

-- the chained_row tables must already exist. The script, utlchain.sql, for creating
   the table is located in /u01/app/oracle/product/12.2.0/dbhome_1/rdbms/admin

----------------------------------- 1) Install Oracle Software --------------------------

NOTE*** You cannot change the Oracle home user after software installation. To change the user you have to uninstall and re-install Oracle software

1. Download Oracle database and unzip files into chosen Oracle home location (e.g 'C:\App\Oracle\product\19.0.0\db_home\')
2. run the installer (setup.exe) from the Oracle files directory (you can run command "C:\App\Oracle\product\19.0.0\db_home\setup" in cmd prompt)

To uninstall Oracle Software, run the deinstall file from the Oracle home that you want to uninstall

run the command "C:\App\Oracle\product\19.0.0\db_home\deinstall"

----------------------------------- 3) Create a password file -----------------------------------

-- PASSWORD is the password for the SYS user
-- ENTRIES sets the maximum number of high level users that will be saved in the password file
   this number cannot be changed once created

ORAPWD file='C:\App\Oracle\product\19.0.0\db_home\PWDDBName.ora' ENTRIES=15 PASSWORD=your_password_for_sys_user

----------------------------------- 4) Create a new instance -----------------------------------

-- For a complete listing of all the options visit the website below
https://docs.oracle.com/en/database/oracle/oracle-database/19/ntqrf/creating-an-instance-using-oradim.html#GUID-BFE19571-26EE-41D2-B7E6-03C98C7C4452

-- NEW indicates it's a new instance
-- SID is the name for the instance to be created
-- STARTMODE AUTO indicates that the instance should startup up and the database should open
   when the Oracle database service starts (Default is STARTMODE manual)
-- SRVCSTART system indicates that the Oracle database service should start 
   automatically when the sytem boots or reboots (the default is SRVCSTART demand)
-- PFILE filname is the initialization parameter file to be used with the instance.
   The complete path and filename is required
-- SHUTMODE specifies how to stop the instance when the Oracle service is shut down
   The default is SHUTMODE immediate 

ORADIM -NEW -SID CLONEDB -STARTMODE auto -pfile 'location\file_name' -SRVCSTART system -SHUTMODE immediate

-- To edit an already created instance use EDIT in place of NEW and specify the changes
   in the same way as you would for a new instance

ORADIM -EDIT -SID CLONEDB -STARTMODE manual 

----------------------------------- 5) Create an SPFile from the pfile and Startup Instance  -----------------------------------

SET ORACLE_SID=InstanceName

SQLPLUS / AS SYSDBA

CREATE SPFILE FROM PFILE='location\pfile';


----------------------------------- 6) Create the database -----------------------------------

-- With the instance in NOMOUNT mode run the database creation script to create database

CREATE DATABASE ORA12CEX -- 8 character limit for database name
USER SYS IDENTIFIED BY "&&sysPassword"
USER SYSTEM IDENTIFIED BY "&&systemPassword"
-- Multiplexed Redo log files
LOGFILE GROUP 1 ('D:\Oracle\DRASRDW52\Data\redo01.log') SIZE 200M,
	   GROUP 2 ('D:\Oracle\DRASRDW52\Data\redo02.log') SIZE 200M,
	   GROUP 3 ('D:\Oracle\DRASRDW52\Data\redo03.log') SIZE 200M
MAXLOGFILES 5
MAXLOGMEMBERS 5
MAXLOGHISTORY 1
MAXDATAFILES 100
MAXINSTANCES 1
CHARACTER SET AL32UTF8
NATIONAL CHARACTER SET AL16UTF16
EXTENT MANAGEMENT LOCAL
-- Create data files
DATAFILE 'D:\Oracle\DRASRDW52\Data\system01.dbf' 
	SIZE 880M REUSE AUTOEXTEND ON NEXT 200M MAXSIZE UNLIMITED
SYSAUX DATAFILE 'D:\Oracle\DRASRDW52\Data\sysaux01.dbf' 
	SIZE 500M REUSE AUTOEXTEND ON NEXT 100M MAXSIZE UNLIMITED
DEFAULT TABLESPACE users
DATAFILE 'D:\Oracle\DRASRDW52\Data\users01.dbf'
	SIZE 500M REUSE AUTOEXTEND ON NEXT 200M MAXSIZE UNLIMITED
DEFAULT TEMPORARY TABLESPACE TEMP
TEMPFILE 'D:\Oracle\DRASRDW52\Data\temp01.dbf'
	SIZE 100M REUSE AUTOEXTEND ON NEXT 20M MAXSIZE UNLIMITED
UNDO TABLESPACE UNDOTBS
 DATAFILE 'D:\Oracle\DRASRDW52\Data\undotbs01.dbf'
  SIZE 700M REUSE AUTOEXTEND ON NEXT 200M MAXSIZE UNLIMITED;
  
----------------------------------- 7) Install Oracle components  ---------------------------

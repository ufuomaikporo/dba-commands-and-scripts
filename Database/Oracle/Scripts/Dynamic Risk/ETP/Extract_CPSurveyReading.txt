---------------------------------------- CIS DATA --------------------------------
SELECT c."GLOBALID",
	c."CONTRACT",
	c."READINGDATE",
	c."STATICVALUE",
	c."PSON",
	c."PSOFF",
	c."DEPOLPS",
	c."STATUS",
	c."SURVEYTYPE",
	c."MEASURE",
	c."ROUTEID"
FROM CORROSIONCONTROL.CISREADINGS_EVW c
INNER JOIN ATLAS.CIS_PREVCURR curr
	ON c.CONTRACT = curr.CIS_CUR_GLOBALID
		AND c.ROUTEID = curr.ROUTEID
WHERE c.MEASURE BETWEEN curr.FROM_MEASURE AND curr.TO_MEASURE
	AND (
		c.PSON <> 0
		OR c.PSOFF <> 0
		)

	DATASOURCE,
	STATUS
FROM ATLAS.STATIONEQUATION se
WHERE UPPER(se.DATASOURCE) IN ('TRANSMISSION', 'GATHERING', 'DETAIL PIPE')
	AND UPPER(se.STATUS) IN ('IN-SERVICE', 'IN-SERVICE(IDLE)', 'IN-SERVICE(DEFERRED)', 'ACTIVE')


-------------------------------------- CIS Contract -----------------------------

SELECT GLOBALID,
	FROMROUTEID,
	TOROUTEID,
	FROMMEASURE,
	TOMEASURE,
	UPPER(TYPE) AS TYPE,
	STARTDATE
FROM INTEGRITY.CONTRACT_EVW
WHERE FROMMEASURE IS NOT NULL
	AND TOMEASURE IS NOT NULL
	AND UPPER(TYPE) IN ('CIS', 'CIS DEPOL/STATIC', 'CIS ON/OFF', 'CIS ON/OFF - FAST CYCLE', 'CIS ON/OFF - SLOW CIS CYCLE')
	AND UPPER(STATUS) IN ('IN-SERVICE', 'IN-SERVICE(IDLE)', 'IN-SERVICE(DEFERRED)', 'ACTIVE')

-------------------------------- ATLAS_DYNSEG ---------------

SELECT CAST(el.OBJECTID AS NUMBER(38, 0)) OBJECTID,
	el.PRESSURESYSTEM_GLOBALID,
	el.FROM_MEASURE,
	el.TO_MEASURE,
	el.STATE,
	el.DURATION,
	el.MINADJUSTEDPRESSURE,
	CASE 
		WHEN TO_CHAR(el.TESTDATE, 'YYYY-MM-DD') > '1900-01-01'
			THEN el.TESTDATE
		ELSE NULL
		END TESTDATE,
	el.TESTMEDIUM,
	el.TESTTYPE,
	CASE 
		WHEN TO_CHAR(el.VTC_DATE_VERIFIED, 'YYYY-MM-DD') > '1900-01-01'
			THEN el.VTC_DATE_VERIFIED
		ELSE NULL
		END VTC_DATE_VERIFIED,
	el.VTC_MATERIAL_VERIFIED,
	el.HCA,
	el.MCA,
	el.CLASS_3_4
FROM ATLAS.ATLAS_DYNSEG_PPL_OPERATOR el
INNER JOIN ATLAS.STATIONEQUATION se
	ON se.ROUTEID = el.PRESSURESYSTEM_GLOBALID
		AND UPPER(se.DATASOURCE) IN ('TRANSMISSION', 'GATHERING', 'DETAIL PIPE')
		AND UPPER(se.STATUS) IN ('IN-SERVICE', 'IN-SERVICE(IDLE)', 'IN-SERVICE(DEFERRED)', 'ACTIVE')
WHERE se.Dot_Operator_Name IS NOT NULL
	AND se.Lineloop IS NOT NULL
	AND se.SystemName IS NOT NULL
	AND se.ProductType IS NOT NULL
	AND se.InterStateIntraState IS NOT NULL
	AND se.From_Measure IS NOT NULL
	AND se.To_Measure IS NOT NULL
	AND el.From_Measure BETWEEN se.From_Measure AND se.To_Measure
	AND el.To_Measure BETWEEN se.From_Measure AND se.To_Measure

SELECT cis.GLOBALID,
	cis.CONTRACT,
	cis.READINGDATE,
	CASE 
		WHEN cis.PSON >= 10
			OR cis.PSON <= - 10
			THEN cis.PSON / 1000
		ELSE cis.PSON
		END AS PSON,
	CASE 
		WHEN cis.PSOFF >= 10
			OR cis.PSOFF <= - 10
			THEN cis.PSOFF / 1000
		ELSE cis.PSOFF
		END AS PSOFF,
	CASE 
		WHEN cis.STATICVALUE >= 10
			OR cis.STATICVALUE <= - 10
			THEN cis.STATICVALUE / 1000
		ELSE cis.STATICVALUE
		END AS STATICVALUE,
	CASE 
		WHEN cis.DEPOLPS >= 10
			OR cis.DEPOLPS <= - 10
			THEN cis.DEPOLPS / 1000
		ELSE cis.DEPOLPS
		END AS DEPOLPS,
	cis.ROUTEID,
	cis.MEASURE
FROM (
	SELECT cis.GLOBALID,
		cis.CONTRACT,
		cis.READINGDATE,
		CASE 
			WHEN cis.PSON = 0
				THEN NULL
			ELSE cis.PSON
			END PSON,
		CASE 
			WHEN cis.PSOFF = 0
				THEN NULL
			ELSE cis.PSOFF
			END PSOFF,
		CASE 
			WHEN cis.STATICVALUE = 0
				THEN NULL
			ELSE cis.STATICVALUE
			END STATICVALUE,
		CASE 
			WHEN cis.DEPOLPS = 0
				THEN NULL
			ELSE cis.DEPOLPS
			END DEPOLPS,
		cis.ROUTEID,
		cis.MEASURE,
		cis.STATUS
	FROM dbo.ETL_PreExtract_CIS_Data cis
	) cis
INNER JOIN dbo.ETL_PreExtract_CIS_Contract c
	ON c.GLOBALID = cis.CONTRACT
INNER JOIN dbo.ETL_PreExtract_Atlas_StationEquation se
	ON se.ROUTEID = cis.ROUTEID
		AND UPPER(se.DataSource) IN ('TRANSMISSION', 'GATHERING', 'DETAIL PIPE')
		AND UPPER(se.STATUS) IN ('IN-SERVICE', 'IN-SERVICE(IDLE)', 'IN-SERVICE(DEFERRED)', 'ACTIVE')
WHERE UPPER(cis.STATUS) IN ('IN-SERVICE', 'IN-SERVICE(IDLE)', 'IN-SERVICE(DEFERRED)', 'ACTIVE')
	AND c.TYPE IN ('CIS', 'CIS DEPOL/STATIC')

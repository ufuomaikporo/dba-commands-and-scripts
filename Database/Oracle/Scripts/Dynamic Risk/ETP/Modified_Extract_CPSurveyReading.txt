SELECT cis.GLOBALID,
	cis.CONTRACT,
	cis.READINGDATE,
	CASE 
		WHEN cis.PSON >= 10
			OR cis.PSON <= - 10
			THEN cis.PSON / 1000
		ELSE cis.PSON
		END AS PSON,
	CASE 
		WHEN cis.PSOFF >= 10
			OR cis.PSOFF <= - 10
			THEN cis.PSOFF / 1000
		ELSE cis.PSOFF
		END AS PSOFF,
	CASE 
		WHEN cis.STATICVALUE >= 10
			OR cis.STATICVALUE <= - 10
			THEN cis.STATICVALUE / 1000
		ELSE cis.STATICVALUE
		END AS STATICVALUE,
	CASE 
		WHEN cis.DEPOLPS >= 10
			OR cis.DEPOLPS <= - 10
			THEN cis.DEPOLPS / 1000
		ELSE cis.DEPOLPS
		END AS DEPOLPS,
	cis.ROUTEID,
	cis.MEASURE
FROM (
	SELECT cis.GLOBALID,
		cis.CONTRACT,
		cis.READINGDATE,
		CASE 
			WHEN cis.PSON = 0
				THEN NULL
			ELSE cis.PSON
			END PSON,
		CASE 
			WHEN cis.PSOFF = 0
				THEN NULL
			ELSE cis.PSOFF
			END PSOFF,
		CASE 
			WHEN cis.STATICVALUE = 0
				THEN NULL
			ELSE cis.STATICVALUE
			END STATICVALUE,
		CASE 
			WHEN cis.DEPOLPS = 0
				THEN NULL
			ELSE cis.DEPOLPS
			END DEPOLPS,
		cis.ROUTEID,
		cis.MEASURE,
		cis.STATUS
	FROM CORROSIONCONTROL.CISREADINGS_EVW cis
	) cis
INNER JOIN (SELECT GLOBALID,
	FROMROUTEID,
	TOROUTEID,
	FROMMEASURE,
	TOMEASURE,
	UPPER(TYPE) AS TYPE,
	STARTDATE
FROM INTEGRITY.CONTRACT_EVW
WHERE FROMMEASURE IS NOT NULL
	AND TOMEASURE IS NOT NULL
	AND UPPER(TYPE) IN ('CIS', 'CIS DEPOL/STATIC', 'CIS ON/OFF', 'CIS ON/OFF - FAST CYCLE', 'CIS ON/OFF - SLOW CIS CYCLE')
	AND UPPER(STATUS) IN ('IN-SERVICE', 'IN-SERVICE(IDLE)', 'IN-SERVICE(DEFERRED)', 'ACTIVE')
) c
	ON c.GLOBALID = cis.CONTRACT
INNER JOIN ATLAS.STATIONEQUATION se
	ON se.ROUTEID = cis.ROUTEID
		AND UPPER(se.DataSource) IN ('TRANSMISSION', 'GATHERING', 'DETAIL PIPE')
		AND UPPER(se.STATUS) IN ('IN-SERVICE', 'IN-SERVICE(IDLE)', 'IN-SERVICE(DEFERRED)', 'ACTIVE')
WHERE UPPER(cis.STATUS) IN ('IN-SERVICE', 'IN-SERVICE(IDLE)', 'IN-SERVICE(DEFERRED)', 'ACTIVE')
	AND c.TYPE IN ('CIS', 'CIS DEPOL/STATIC')
	OR 
	(c.TYPE IN ('CIS ON/OFF', 'CIS ON/OFF - FAST CYCLE', 'CIS ON/OFF - SLOW CIS CYCLE') AND cis.PSOFF <> 0)

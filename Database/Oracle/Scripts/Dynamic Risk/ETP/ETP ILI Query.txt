DROP TABLE Temp_ILIFeature;

CREATE TABLE Temp_ILIFeature AS
SELECT C.GlobalID,
	UPPER(C.CompanyId) AS CompanyId,
	'UNKNOWN' AS ILICallBoxInteractRule,
	CASE 
		WHEN c.STARTDATE IS NULL
			AND c.PLANNEDDATE IS NULL
			THEN SYSDATE
		WHEN c.STARTDATE IS NULL
			THEN c.PLANNEDDATE
		ELSE c.STARTDATE
		END AS STRTDATE_PLNDDATE,
	CASE 
		WHEN c.FINISHDATE IS NULL
			AND c.PLANNEDDATE IS NULL
			THEN SYSDATE
		WHEN c.FINISHDATE IS NULL
			THEN c.PLANNEDDATE
		ELSE c.FINISHDATE
		END AS FNSHDATE_PLNDDATE,
	UPPER(C.STATUS) AS STATUS,
	UPPER(c.TYPE) AS TYPE,
    	UPPER(NVL(SUBSTR(C.TOOL_TECHNOLOGY, 0, INSTR(C.TOOL_TECHNOLOGY, ':')-1), C.TOOL_TECHNOLOGY)) AS TOOL_TECHNOLOGY,
	CASE 
		WHEN C.CONTRACTNUMBER IS NULL
			THEN CAST(c.OBJECTID AS VARCHAR2(20))
		ELSE C.CONTRACTNUMBER
		END AS CONTRACTNUMBER,
	CASE 
		WHEN c.BEGWHEELCNT IS NULL
			THEN c.FROMMEASURE
		WHEN c.BEGWHEELCNT > 9999999
			THEN 9999999
		ELSE c.BEGWHEELCNT
		END AS BEGWHLCNT_FRMMSR,
	CASE 
		WHEN c.ENDWHEELCNT IS NULL
			THEN c.TOMEASURE
		WHEN c.ENDWHEELCNT > 9999999
			THEN 9999999
		ELSE C.ENDWHEELCNT
		END AS ENDWHLCNT_TOMSR,
	C.FROMROUTEID AS FROMROUTEID,
	C.FROMMEASURE,
	C.TOROUTEID AS TOROUTEID,
	C.TOMEASURE
FROM Atlas.Contract C
INNER JOIN ATLAS.STATIONEQUATION SEFR
	ON SEFR.ROUTEID = C.FROMROUTEID
		AND UPPER(SEFR.DataSource) IN ('TRANSMISSION', 'GATHERING', 'DETAIL PIPE')
		AND UPPER(SEFR.STATUS) IN ('IN-SERVICE', 'IN-SERVICE(IDLE)', 'IN-SERVICE(DEFERRED)', 'ACTIVE')
INNER JOIN ATLAS.STATIONEQUATION SETR
	ON SETR.ROUTEID = C.TOROUTEID
		AND UPPER(SETR.DataSource) IN ('TRANSMISSION', 'GATHERING', 'DETAIL PIPE')
		AND UPPER(SETR.STATUS) IN ('IN-SERVICE', 'IN-SERVICE(IDLE)', 'IN-SERVICE(DEFERRED)', 'ACTIVE')
WHERE SEFR.Dot_Operator_Name IS NOT NULL
	AND SEFR.Lineloop IS NOT NULL
	AND SEFR.SystemName IS NOT NULL
	AND SEFR.ProductType IS NOT NULL
	AND SEFR.InterStateIntraState IS NOT NULL
	AND SEFR.From_Measure IS NOT NULL
	AND SEFR.To_Measure IS NOT NULL
	--AND (
	--	Upper(C.Type) LIKE '%MFL%'
	--	OR Upper(C.Type) LIKE '%GEO%'
	--	OR Upper(C.Type) LIKE 'UT%'
	--	OR C.Type IN ('HR-Combo', 'EMAT', 'Low Res Caliper')
	--	)
	AND Upper(C.Type) = 'ILI'
	AND UPPER(C.STATUS) IN ('IN-SERVICE', 'IN-SERVICE(IDLE)', 'IN-SERVICE(DEFERRED)', 'ACTIVE')
	AND C.FROMMEASURE BETWEEN SEFR.from_measure AND SEFR.to_measure
	AND C.toMEASURE BETWEEN SETR.from_measure AND SETR.to_measure;

SELECT GLOBALID
       ,COMPANYID
       ,ILICallBoxInteractRule
       ,STRTDATE_PLNDDATE
       ,FNSHDATE_PLNDDATE
       ,STATUS
       ,Type
       ,TRIM(REGEXP_SUBSTR (
       REGEXP_REPLACE(Tool_Technology, '[^A-Z,a-z, ,;,-]| - '),
       '[^;]+',
       1,
       LEVEL
     )) Tool_Technology
     ,SUBSTR(TRIM(REGEXP_SUBSTR(REGEXP_REPLACE(Tool_Technology, '[^0-9,/, ,;]'),
     '[^;]+',
     1,
     LEVEL
     )), 1, 10) ToolStartDate
     ,TRIM(SUBSTR(TRIM(REGEXP_SUBSTR(REGEXP_REPLACE(Tool_Technology, '[^0-9,/, ,;]'),
     '[^;]+',
     1,
     LEVEL
     )), INSTR(TRIM(REGEXP_SUBSTR(REGEXP_REPLACE(Tool_Technology, '[^0-9,/, ,;]'),
     '[^;]+',
     1,
     LEVEL
     )),' '), 20)) ToolEndDate
     ,contractnumber
     ,BEGWHLCNT_FRMMSR
     ,ENDWHLCNT_TOMSR
     ,FROMROUTEID
     ,FROMMEASURE
     ,TOROUTEID
     ,TOMEASURE
FROM Temp_ILIFeature
CONNECT BY LEVEL <= 
LENGTH (Tool_Technology) - LENGTH (REGEXP_REPLACE(Tool_Technology, ';' )) + 1;

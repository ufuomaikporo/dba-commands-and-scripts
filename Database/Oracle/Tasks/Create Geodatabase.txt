#########################################################################################################
# PRIVILEGES
#########################################################################################################

- The error ORA-06598: insufficient INHERIT PRIVILEGES privilege occurs in Oracle Database 12c and later when a more powerful user
calls a function or procedure owned by a less powerful user, and that function is set to run with the caller's privileges (AUTHID CURRENT_USER)

- a user with DBA privileges needs to explicitly grant the INHERIT PRIVILEGES privilege to the SDE schema on the calling user's account, 
 thereby indicating that the calling user trusts the code within the SDE schema
 
- If the calling user is SYS, the grant command would be

GRANT INHERIT PRIVILEGES ON USER SYS TO SDE;

-- Check which user accounts have INHERIT PRIVILEGES on the SDE schema. This means there is trust with SDE schema to execute functions
 within it that use the caller's privileges (if the caller is SYS, then SYS user privileges are used to execute the function)

SELECT *
FROM DBA_TAB_PRIVS
WHERE PRIVILEGE = 'INHERIT PRIVILEGES'
AND TABLE_NAME = 'SDE';

# ADD A NEW GEODATABASE SCHEMA/USER AND IMPORT GEODATABASE OBJECTS
#########################################################################################################

STEP 1) Create a new tablespace for the new schema

CREATE BIGFILE TABLESPACE New_Schema 
DATAFILE 'C:\app\oracle\oradata\ORCL\New_Schema.DBF' SIZE 200m REUSE
AUTOEXTEND ON NEXT 100M MAXSIZE 4GB LOGGING 
EXTENT MANAGEMENT LOCAL 
SEGMENT SPACE MANAGEMENT AUTO;

STEP 2) Create the new schema user

CREATE USER new_schema_user IDENTIFIED BY new_user
DEFAULT TABLESPACE New_Schema
QUOTA UNLIMITED ON New_Schema
PROFILE DEFAULT;

STEP 3) Open ArcGIS OR ArcCatalog app and 'Create a New Database Connection' using the new schema username and password.
		You can rename the connection after it is created 

STEP 4) 

OPTION 1) Copy and paste the tables and feature classes from the gdb file to the geodatabase connection file (faster option)

OPTION 2) Use the import feature in ESRI
		
		i) For tables, right click the new geodatabase connection then click import > Table(s). A geoprocessing tab will pop-up on the right.
			Use the 'Type' column to sort by type so all 'Enterprise Geodatabase Table' can be highlighted at once (Ctrl + Shift + down arrow). 
			Then select the geodatabase connection created earlier for the 'Output Geodatabase' parameter then click run
			
		ii) For feature classes, right click the new geodatabase connection then click import > Feature(s). A geoprocessing tab will pop-up on the right. 
			Use the 'Type' column to sort by type so all 'Enterprise Geodatabase Feature Class' items can be highlighted at once (Ctrl + Shift + down arrow). 
			Then select the geodatabase connection created earlier for the 'Output Geodatabase' parameter then click run
		
################################################################################################################
DELETING GEODATABASE TABLESPACE
##############################################################################################################

-- For Example if we want to drop a geodatabase schema called 'ATCO_UPDM_PUB' 

--############# STEP 1) Find Secondary domain indexes

SET PAGES 10000
SET LIN 150
COL owner FORMAT A20
COL index_name FORMAT A30
COL table_name FORMAT A25
COL table_owner FORMAT A20
COL tablespace_name FORMAT A25

SELECT owner, index_name, table_owner, table_name, tablespace_name 
FROM all_indexes 
WHERE index_type='DOMAIN' AND table_owner='ATCO_UPDM_PUB';

--################ STEP 2) Drop Secondary objects Domain indexes

SELECT 'DROP INDEX ' || table_owner || '.' || index_name || ';'
FROM all_indexes
WHERE index_type='DOMAIN' AND table_owner='ATCO_UPDM_PUB';

SELECT 'DROP MATERIALIZED VIEW ' || owner || '.' || mview_name || ';'
FROM dba_mviews
WHERE owner = 'ETP_ATLAS_PROD';

--####################### STEP 3) Drop as many objects or materialized views as needed for tablespace drop to work

DROP TABLESPACE ATCO_UPDM_PUB INCLUDING CONTENTS AND DATAFILES;

# DROPPNG THE SDE TABLESPACE
###################################################################################################

-- The SDE tablespace is the main geodatabase tablespace in an Oracle database. It contains info being 
used by other geodatabase schemas/tablespaces. So before dropping the SDE tablespace you should drop all 
Geodatabse tablespaces first

--############# STEP 1) Find Secondary domain indexes

SET PAGES 10000
SET LIN 150
COL owner FORMAT A20
COL index_name FORMAT A30
COL table_name FORMAT A25
COL table_owner FORMAT A20
COL tablespace_name FORMAT A25

SELECT owner, index_name, table_owner, table_name, tablespace_name 
FROM all_indexes 
WHERE index_type='DOMAIN' AND table_owner='SDE';

--################ STEP 2) Drop Secondary Domain indexes

SELECT 'DROP INDEX ' || table_owner || '.' || index_name || ';'
FROM all_indexes
WHERE index_type='DOMAIN' AND table_owner='SDE';

--####################### STEP 3) Drop as many objects or materialized views as needed for tablespace drop to work (MAY NOT NEED TO DO THIS)

SELECT 'DROP ' || object_type || ' ' || owner || '.' || object_name || ';'
FROM dba_objects
WHERE owner = 'SDE';

SELECT 'DROP MATERIALIZED VIEW ' || owner || '.' || mview_name || ';'
FROM dba_mviews
WHERE owner = 'SDE';

--###################### STEP 4) Drop tablespace

DROP TABLESPACE ATCO_UPDM_PUB INCLUDING CONTENTS AND DATAFILES;

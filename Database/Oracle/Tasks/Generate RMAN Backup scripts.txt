REPLACE SCRIPT Weekly_Full_Backup
{
	ALLOCATE CHANNEL c1 DEVICE TYPE DISK FORMAT 'B:\Backups\RMAN_Testing\%d_%I_%M_%D_%Y_%U' MAXPIECESIZE 3G;
	ALLOCATE CHANNEL c2 DEVICE TYPE DISK FORMAT 'B:\Backups\RMAN_Testing\Test\%d_%I_%M_%D_%Y_%U' MAXPIECESIZE 3G;
	ALLOCATE CHANNEL c3 DEVICE TYPE DISK FORMAT 'B:\Backups\RMAN_Testing\%d_%I_%M_%D_%Y_%U' MAXPIECESIZE 3G;
	ALLOCATE CHANNEL c4 DEVICE TYPE DISK FORMAT 'B:\Backups\RMAN_Testing\%d_%I_%M_%D_%Y_%U' MAXPIECESIZE 3G;

	BACKUP AS COMPRESSED BACKUPSET 
	INCREMENTAL LEVEL 0 
	CHECK LOGICAL 
	TAG='FULL_Weekly_Backup'
	DATABASE
	PLUS ARCHIVELOG
	DELETE ALL INPUT;

	RELEASE CHANNEL c1 ;
	RELEASE CHANNEL c2 ;
	RELEASE CHANNEL c3 ;
	RELEASE CHANNEL c4 ;
}
